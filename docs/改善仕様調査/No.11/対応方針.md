## 対応3: フロントエンドエラーユーティリティ関数の実装

### 対象ファイル
- **新規作成**: `frontend/src/utils/errorHandler.ts`

### 対応内容

reference版 `errorHandler.js` を TypeScript 化して移植する。  
`ERROR_CODES`（対応2で作成）をimportして使用する。

```typescript
// frontend/src/utils/errorHandler.ts

import { ERROR_CODES, ErrorDefinition } from './errorCodes';

/**
 * エラー情報オブジェクト（アプリケーション内部で使用）
 */
export interface AppError {
  code: string;
  category: string;
  message: string;
  detail: string;
  solution: string;
  timestamp: string;
  originalError: Error | null;
}

/**
 * エラーコードからエラー情報オブジェクトを生成
 */
export function createError(
  errorCode: string,
  options: {
    detail?: string;
    solution?: string;
    originalError?: Error;
  } = {}
): AppError {
  const errorDef: ErrorDefinition = ERROR_CODES[errorCode] ?? ERROR_CODES['E9999'];
  return {
    code: errorDef.code,
    category: errorDef.category,
    message: errorDef.message,
    detail: options.detail ?? errorDef.detail,
    solution: options.solution ?? errorDef.solution,
    timestamp: new Date().toISOString(),
    originalError: options.originalError ?? null,
  };
}

/**
 * HTTPステータスコードからエラーコードを判定
 */
export function getErrorCodeFromStatus(
  statusCode: number,
  errorMessage: string = ''
): string {
  if (statusCode === 401) return 'E5001';
  if (statusCode === 403) return 'E5002';
  if (statusCode >= 400 && statusCode < 500) {
    if (errorMessage.includes('validation')) return 'E1003';
    if (errorMessage.includes('required'))   return 'E1004';
    return 'E1002';
  }
  if (statusCode >= 500) {
    if (errorMessage.includes('database')) return 'E4002';
    return 'E4001';
  }
  return 'E9999';
}

/**
 * fetch レスポンスのエラーを処理
 */
export async function handleFetchError(
  error: Error,
  response: Response | null = null
): Promise<AppError> {
  if (!response) {
    if (error.message.includes('timeout')) {
      return createError('E3002', { originalError: error });
    }
    return createError('E3001', { originalError: error });
  }
  let errorMessage = '';
  try {
    const data = await response.json();
    errorMessage = data.error?.message || data.message || '';
  } catch {
    errorMessage = response.statusText;
  }
  const errorCode = getErrorCodeFromStatus(response.status, errorMessage);
  return createError(errorCode, { detail: errorMessage, originalError: error });
}

/**
 * ファイルアップロードエラーを処理
 */
export function handleFileUploadError(file: File | null, error: Error): AppError {
  const MAX_FILE_SIZE = 50 * 1024 * 1024; // 50MB
  if (file && file.size > MAX_FILE_SIZE) {
    return createError('E1001', {
      detail: `ファイルサイズ: ${(file.size / 1024 / 1024).toFixed(2)}MB（上限: 50MB）`,
      originalError: error,
    });
  }
  if (file && !file.name.endsWith('.maiml')) {
    return createError('E1002', {
      detail: `ファイル形式: .${file.name.split('.').pop()}`,
      originalError: error,
    });
  }
  return createError('E1001', { originalError: error });
}

/**
 * XMLパースエラーを処理
 */
export function handleXMLParseError(error: Error, _xmlContent: string = ''): AppError {
  let detail = 'XML形式が不正です';
  const lineMatch = error.message.match(/line (\d+)/i);
  if (lineMatch) {
    detail = `XML形式が不正です（行${lineMatch[1]}）`;
  }
  return createError('E2001', { detail, originalError: error });
}

/**
 * エラーをコンソールにログ出力（[Exxx] 形式でコードを含める）
 */
export function logError(errorInfo: AppError): void {
  console.error(`[${errorInfo.code}] ${errorInfo.message}`, {
    category: errorInfo.category,
    detail: errorInfo.detail,
    solution: errorInfo.solution,
    timestamp: errorInfo.timestamp,
    originalError: errorInfo.originalError,
  });
}

/**
 * 任意のエラーオブジェクトを AppError に統一
 */
export function normalizeError(error: unknown): AppError {
  if (error instanceof Error) {
    return createError('E9999', { detail: error.message, originalError: error });
  }
  if (typeof error === 'string') {
    return createError('E9999', { detail: error });
  }
  return createError('E9999');
}
```

```typescript
interface LogEntry {

  level: 'log' | 'info' | 'warn' | 'error';
  message: string;
  url: string;
  userAgent: string;
  stackTrace?: string;
}

const MAX_LOGS = 500;

class ConsoleLogger {
  private logs: LogEntry[] = [];
  private sessionId: string;
  private isCapturing = false;
  private readonly originalConsole = {
    log:   console.log.bind(console),
    info:  console.info.bind(console),
    warn:  console.warn.bind(console),
    error: console.error.bind(console),
  };

  constructor() {
    this.sessionId = `session_${Date.now()}_${Math.random().toString(36).substring(7)}`;
  }

  /** ログキャプチャ開始 */
  startCapture(): void {
    if (this.isCapturing) return;
    this.isCapturing = true;

    (['log', 'info', 'warn', 'error'] as const).forEach((level) => {
      (console as any)[level] = (...args: unknown[]) => {
        this.captureLog(level, args);
        this.originalConsole[level](...args);
      };
    });
  }

  /** ログキャプチャ停止 */
  stopCapture(): void {
    if (!this.isCapturing) return;
    console.log   = this.originalConsole.log;
    console.info  = this.originalConsole.info;
    console.warn  = this.originalConsole.warn;
    console.error = this.originalConsole.error;
    this.isCapturing = false;
  }

  /** ログを内部バッファに保存 */
  private captureLog(level: LogEntry['level'], args: unknown[]): void {
    try {
      const message = args
        .map((arg) => {
          if (typeof arg === 'object') {
            try { return JSON.stringify(arg); } catch { return String(arg); }
          }
          return String(arg);
        })
        .join(' ');

      const entry: LogEntry = {
        timestamp: new Date().toISOString(),
        level,
        message,
        url: window.location.href,
        userAgent: navigator.userAgent,
      };

      if (level === 'error' && args[0] instanceof Error) {
        entry.stackTrace = (args[0] as Error).stack;
      }

      this.logs.push(entry);
      if (this.logs.length > MAX_LOGS) {
        this.logs.shift();
      }
    } catch {
      // キャプチャ中のエラーは無視
    }
  }

  getLogs(): LogEntry[] { return this.logs; }
  clearLogs(): void     { this.logs = []; }

  /** バックエンドにログを送信 */
  async sendLogsToBackend(userId?: string): Promise<{ success: boolean; count?: number }> {
    try {
      const res = await axios.post(
        '/api/logs/frontend',
        { logs: this.logs, sessionId: this.sessionId, userId },
        { withCredentials: true }
      );
      this.originalConsole.log(
        `✅ [ConsoleLogger] ログ送信成功: ${res.data.count}件`
      );
      return { success: true, count: res.data.count };
    } catch (error) {
      this.originalConsole.error('❌ [ConsoleLogger] ログ送信エラー:', error);
      return { success: false };
    }
  }

  getStats() {
    return {
      total: this.logs.length,
      byLevel: {
        log:   this.logs.filter((l) => l.level === 'log').length,
        info:  this.logs.filter((l) => l.level === 'info').length,
        warn:  this.logs.filter((l) => l.level === 'warn').length,
        error: this.logs.filter((l) => l.level === 'error').length,
      },
    };
  }
}

// シングルトン
const consoleLogger = new ConsoleLogger();
export default consoleLogger;
```

### ConsoleLogger の呼び出し方法

アプリケーション起動時（`_app.tsx` 等）でキャプチャを開始し、  
ユーザーのエラー報告時やページ遷移時にバックエンドへ送信する。

```typescript
// 例: キャプチャ開始（アプリ起動時）
import consoleLogger from '@/utils/consoleLogger';
consoleLogger.startCapture();

// 例: エラー発生時に送信
await consoleLogger.sendLogsToBackend(userId);
```

## 注意事項

### バックエンドAPIの`/api/logs/frontend`の存在確認

reference版の `backend/reference/routes/logsRoutes.js` にAPIが定義されているが、  
現行版（`backend/src/`）にも同等のAPIが存在するか確認が必要。  
→ 対応4（ConsoleLogger）の実装前に、`backend/src/routes/` 配下を確認すること。

### 既存エラーコード文字列との後方互換性

対応1でコントローラーを修正する際、エラーコードの文字列値は変更しない。  
（変数名は `ERROR_CODES.SYSTEM_ERROR` になるが、値は `'SYSTEM_ERROR'` のまま）  
→ フロントエンドの `errorMapper.ts` との互換性が保たれる。
