export const fetchTemplateFiles = async (

  const response = await apiClient.get(
    `/api/v1/templates/list?organization_id=${organizationId}&category=${category}`

export const fetchTemplateContent = async (
  organizationId: string,
  filePath: string
): Promise<string> => {

  );
  return response.data.content;
};

### プロパティの衝突解決

```typescript
// 既存キーと衝突する場合の処理
const mergeProperties = (
  existing: Record<string, PropertyData>,
  imported: Record<string, PropertyData>
): Record<string, PropertyData> => {
  const result = { ...existing };
  
  Object.entries(imported).forEach(([key, value]) => {
    if (result[key]) {
      // 衝突: ユーザーに確認
      Modal.confirm({
        title: `キー「${key}」は既に存在します`,
        content: '上書きしますか？',
        onOk: () => { result[key] = value; }
      });
    } else {
      // 衝突なし: そのまま追加
      result[key] = value;
    }
  });
  
  return result;
};
```
