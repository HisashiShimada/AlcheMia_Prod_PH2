#### EditorState（エディタ状態）
```typescript
export interface EditorState {
  nodes: Node[];      // React Flowのノード配列
  edges: Edge[];      // React Flowのエッジ配列
}
```

#### History（履歴構造）
```typescript
export interface History {
  past: EditorState[];      // 過去の状態配列
  present: EditorState;     // 現在の状態
  future: EditorState[];    // やり直し可能な状態配列
}
```

#### UseHistoryReturn（返り値）
```typescript
export interface UseHistoryReturn {
  history: History;
  undo: () => void;
  redo: () => void;
  canUndo: boolean;
  canRedo: boolean;
  pushState: (newState: EditorState) => void;
  reset: (newState: EditorState) => void;
}
```
