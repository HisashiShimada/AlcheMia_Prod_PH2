### 1. **型定義の階層化**

```typescript
// 基本型

// 構造型（再帰）
interface S3Folder {
  children: Record<string, S3Folder>;  // 再帰的
}

// オプション型
interface UseNamespaceOperationsOptions { ... }

// 返り値型
interface UseNamespaceOperationsReturn { ... }
```

### 2. **useCallbackの適切な使用**

```typescript
const saveDraft = useCallback(async (draftData: DraftSaveData) => {
  // 実装
}, [userId, fetchDrafts]);  // 依存配列を正確に
```

### 3. **エラーハンドリングの統一**

```typescript
try {
  // 処理
} catch (error) {
  const errorMessage = error instanceof Error ? error.message : 'Unknown error';
  console.error('[フック名] エラー:', error);
  throw new Error(`詳細メッセージ: ${errorMessage}`);
}
```

### 4. **キャッシュ機能の実装**

```typescript
const [cache, setCache] = useState<Record<string, Data>>({});

if (cache[key]) {
  return cache[key];
}

const data = await fetchData(key);
setCache(prev => ({ ...prev, [key]: data }));
```
