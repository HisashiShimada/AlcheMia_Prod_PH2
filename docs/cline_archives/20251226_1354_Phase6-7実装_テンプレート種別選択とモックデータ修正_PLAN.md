### 5.3. デフォルトXMLテンプレート生成

```typescript
const generateDefaultXmlContent = (
  templateType: 'material' | 'condition' | 'result',
  fileName: string
): string => {
  const timestamp = new Date().toISOString();
  const uuid = crypto.randomUUID();
  
  const baseTemplate = `<?xml version="1.0" encoding="UTF-8"?>
<maiml xmlns="http://www.maiml.org/maiml" 
       xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
       xsi:schemaLocation="http://www.maiml.org/maiml maiml.xsd">
  <document>
    <uuid>urn:uuid:${uuid}</uuid>
    <name>${fileName}</name>
    <createdAt>${timestamp}</createdAt>
    <templates>`;

  let templateContent = '';
  
  switch (templateType) {
    case 'material':
      templateContent = `
      <MaterialTemplate>
        <uuid>urn:uuid:${crypto.randomUUID()}</uuid>
        <name>${fileName} Material Template</name>
        <description>Material template for ${fileName}</description>
        <properties>
          <!-- Add material properties here -->
        </properties>
      </MaterialTemplate>`;
      break;
      
    case 'condition':
      templateContent = `
      <ConditionTemplate>
        <uuid>urn:uuid:${crypto.randomUUID()}</uuid>
        <name>${fileName} Condition Template</name>
        <description>Condition template for ${fileName}</description>
        <properties>
          <!-- Add condition properties here -->
        </properties>
      </ConditionTemplate>`;
      break;
      
    case 'result':
      templateContent = `
      <ResultTemplate>
        <uuid>urn:uuid:${crypto.randomUUID()}</uuid>
        <name>${fileName} Result Template</name>
        <description>Result template for ${fileName}</description>
        <properties>
          <!-- Add result properties here -->
        </properties>
        <contents>
          <!-- Add result contents here -->
        </contents>
      </ResultTemplate>`;
      break;
  }

  return baseTemplate + templateContent + `
    </templates>
  </document>
</maiml>`;
};
```
