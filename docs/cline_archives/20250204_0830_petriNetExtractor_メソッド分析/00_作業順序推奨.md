### ステップ1: ParsedProperty型の定義
```typescript
// frontend/src/utils/petriNetExtractor.ts の先頭に追加
type ParsedProperty = {
  type?: string;
  xsiType?: string;
  key?: string;
  name?: string;
  tagName?: string;
  elements?: {
    uri?: { defaultValue?: string };
    hash?: { defaultValue?: string };
    format?: { defaultValue?: string };
    value?: { defaultValue?: any };
    description?: { defaultValue?: string };
    [key: string]: any;
  };
  defaultValue?: any;
  description?: string;
  units?: string;
  formatString?: string;
  scaleFactor?: number;
  properties?: ParsedProperty[];
  [key: string]: any;
};
```
