### 修正後のコード

#### ルート定義（projectRoutes.ts）
```typescript
router.post(
  '/:projectId/approval-request',
  authMiddleware,
  tenantIsolation,
  [
    param('projectId')
      .isUUID()
      .withMessage('projectIdはUUID形式である必要があります'),
    body('comment')
      .optional()
      .isLength({ max: 1000 })
      .withMessage('commentは1000文字以内である必要があります'),
  ],
  handleValidationErrors,  // ← 追加
  projectController.requestApproval
);
```

#### コントローラー（projectController.ts）
```typescript
export async function requestApproval(req: Request, res: Response): Promise<void> {
  const requestId = uuidv4();
  const projectService = new ProjectService();

  try {
    // バリデーションはミドルウェアで処理済み
    const { organizationId, userId } = req.user!;
    const { projectId } = req.params;
    const { comment } = req.body;

    // Service層呼び出し
    const result = await projectService.requestApproval(
      projectId,
      comment || null,
      organizationId,
      userId
    );

    // エラーハンドリング（ビジネスロジックエラーのみ）
    if (!result.success) {
      // ... 省略 ...
    }

    // 成功レスポンス
    res.status(200).json({
      success: true,
      data: result.data,
      meta: {
        timestamp: new Date().toISOString(),
        requestId,
      },
    });
  } catch (error) {
    // ... エラーハンドリング ...
  }
}
```
