## 5. 技術的詳細

### フロントエンド・バックエンドのデータマッピング

#### フロントエンド (CreateProjectModal) - 統一後
```typescript
interface CreateProjectData {
  projectName: string;  // ← nameから変更（バックエンドと統一）
  description: string;
  folderName: string;   // 自動生成
  owner: string;        // currentUserから取得
  department: string;   // currentUserから取得
  organizationId: string; // currentUserから取得
}
```

#### バックエンドAPI (POST /api/v1/projects)
```typescript
// リクエストボディ
{
  projectName: string;  // ← name
  description?: string; // ← description
}

// レスポンス (201 Created)
{
  success: true;
  data: {
    projectId: string;
    projectName: string;
    description: string;
    status: string;
    organizationId: string;
    departmentId: string;
    createdBy: string;
    createdAt: string;
  };
  meta: {
    timestamp: string;
    requestId: string;
  };
}
```

**注意点:**
- フロントエンドとバックエンドで`projectName`に統一（従来の`name`から変更）
- `folderName`, `owner`, `department`はフロントエンドでのみ使用（バックエンドには送信不要）
- 認証情報（organizationId, userId, departmentId）はJWTトークンから自動取得

### API通信関数の実装方針

```typescript
/**
 * プロジェクト作成APIリクエスト型
 */
export interface CreateProjectRequest {
  projectName: string;
  description?: string;
}

/**
 * プロジェクト作成APIレスポンス型
 */
export interface CreateProjectResponse {
  projectId: string;
  projectName: string;
  description: string;
  status: string;
  organizationId: string;
  departmentId: string;
  createdBy: string;
  createdAt: string;
}

/**
 * プロジェクト作成
 */
export const createProject = async (
  request: CreateProjectRequest
): Promise<CreateProjectResponse> => {
  const client = createApiClient();
  
  const response = await client.post<{
    success: true;
    data: CreateProjectResponse;
    meta: { timestamp: string; requestId: string; };
  }>('/projects', request);
  
  return response.data.data;
};
```
