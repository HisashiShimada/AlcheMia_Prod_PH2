### ProjectManagement.tsx の変更内容

#### 変更前
```typescript
import mockProjectsData from '@/data/mock/mockProjects.json';
import mockFilesData from '@/data/mock/mockFiles.json';

const [projects, setProjects] = useState<any[]>([]);

useEffect(() => {
  setProjects(mockProjectsData.projects);
}, []);
```

#### 変更後
```typescript
import { useApi } from '@/hooks/useApi';
import { fetchProjectsMock, fetchProjectFilesMock } from '@/utils/mockApiUtils';
import type { ProjectListResponse } from '@/types/project';
import type { ProjectFilesResponse } from '@/types/file';
import { Spin, Alert } from 'antd';

// プロジェクト一覧取得
const { 
  data: projectsData, 
  isLoading: isLoadingProjects, 
  error: projectsError, 
  refetch: refetchProjects 
} = useApi<ApiResponse<ProjectListResponse>>(fetchProjectsMock, []);

const projects = projectsData?.data.projects || [];

// ローディング表示
if (isLoadingProjects) {
  return <Spin size="large" />;
}

// エラー表示
if (projectsError) {
  return <Alert message="エラー" description={projectsError.message} type="error" />;
}
```

```typescript
import { useApi } from '@/hooks/useApi';
import { fetchProjectsMock } from '@/utils/mockApiUtils';

  data: projectsData, 
  isLoading, 
  error, 
  refetch 
} = useApi<ApiResponse<ProjectListResponse>>(fetchProjectsMock, []);

const allData: ProgressRecord[] = useMemo(() => {
  if (!projectsData?.data.projects) return [];
  return projectsData.data.projects.map((item, index) => ({
    // マッピング処理
  }));
}, [projectsData]);
```
