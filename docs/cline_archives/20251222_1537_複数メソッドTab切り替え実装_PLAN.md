```typescript
export function parseMaimlToPetriNet(maimlXml: string, selectedMethodId?: string): PetriNetStructure {
  // XMLドキュメントをパース
  const xmlDoc = parser.parseFromString(maimlXml, 'text/xml');
  
  // 🎯 メソッドフィルタリング
  let targetDoc: Document | Element = xmlDoc;
  if (selectedMethodId) {
    const methodElement = xmlDoc.querySelector(`method[id="${selectedMethodId}"]`);
    if (methodElement) {
      targetDoc = methodElement;  // 指定されたメソッドのみを対象
    }
  }
  
  // targetDocから要素を抽出
  const placeElements = targetDoc.querySelectorAll('place');
  const transitionElements = targetDoc.querySelectorAll('transition');
  const arcElements = targetDoc.querySelectorAll('arc');
  // ...
}
```

## 6. 発生した問題と解決方法

### 問題1: Tab切り替え時に1つ目に戻ってしまう

**原因:**
- useEffectの依存配列に`isMultiMethod`と`methods.length`が含まれていた
- Tab切り替え → selectedMethodIndex更新 → 再レンダリング → useEffect再実行 → `setSelectedMethodIndex(0)` → Tab1に戻る

**解決策:**
1. `checkedFileKey`でファイル単位のチェック済み管理を実装
2. 依存配列を`[fileContent, selectedFile?.key]`のみに変更
3. 既にチェック済みのファイルは早期リターンでスキップ

### 問題2: メッセージが複数回表示される

**原因:**
- useEffectが複数回実行されていた
- ダイアログ初期表示時に2回実行

**解決策:**
1. `checkedFileKey`でファイル単位の実行制御
2. `message.success()`に`key`パラメータを追加して重複防止
3. 依存配列をシンプルにして不要な再実行を防止

### 問題3: PetriNetViewer内の不要な処理

**原因:**
- PetriNetViewer内でtest_omori11検出時に`message.info()`を表示していた
- この処理が不要で混乱を招いていた

**解決策:**
- PetriNetViewer内の不要な処理を削除
- ViewRendererで一元管理
