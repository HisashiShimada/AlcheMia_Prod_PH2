## 5. 実装コード例

### 5.1. 複数method対応（概要）

```typescript
export async function parsePetriNetStructure(content: string): Promise<any> {
  try {
    const parser = new DOMParser();
    const doc = parser.parseFromString(content, 'text/xml');
    
    // Protocol取得
    const protocol = doc.getElementsByTagName('protocol')[0];
    if (!protocol) {
      throw new Error('Protocol element not found');
    }
    
    // Method要素を取得（配列対応）
    const methodElements = protocol.getElementsByTagName('method');
    const methodArray = Array.from(methodElements);
    const isMultiMethod = methodArray.length > 1;
    
    const methods = [];
    const allPlaces = [];
    const allTransitions = [];
    const allArcs = [];
    
    // 各methodを処理
    for (let i = 0; i < methodArray.length; i++) {
      const method = methodArray[i];
      const methodId = method.getAttribute('id') || `method_${i}`;
      
      const methodData = {
        methodId,
        methodIndex: i,
        places: [],
        transitions: [],
        arcs: [],
        metadata: extractMethodMetadata(method),
        statistics: {
          placeCount: 0,
          transitionCount: 0,
          arcCount: 0
        }
      };
      
      // PNML要素から各ノードを抽出
      const pnml = method.getElementsByTagName('pnml')[0];
      if (pnml) {
        methodData.places = extractPlacesFromPnml(pnml, methodId);
        methodData.transitions = extractTransitionsFromPnml(pnml, methodId);
        methodData.arcs = extractArcsFromPnml(pnml, methodId);
        
        methodData.statistics.placeCount = methodData.places.length;
        methodData.statistics.transitionCount = methodData.transitions.length;
        methodData.statistics.arcCount = methodData.arcs.length;
      }
      
      methods.push(methodData);
      allPlaces.push(...methodData.places);
      allTransitions.push(...methodData.transitions);
      allArcs.push(...methodData.arcs);
    }
    
    return {
      success: true,
      isMultiMethod,
      methods,
      // 後方互換性
      places: allPlaces,
      transitions: allTransitions,
      arcs: allArcs,
      statistics: {
        placeCount: allPlaces.length,
        transitionCount: allTransitions.length,
        arcCount: allArcs.length,
        methodCount: methods.length
      }
    };
  } catch (error) {
    console.error('Parse error:', error);
    throw error;
  }
}
```

### 5.2. 重複ID検証（概要）

```typescript
export function validateDuplicateIds(doc: Document): ValidationResult {
  const idMap = new Map<string, Array<{ elementType: string; path: string }>>();
  const errors: string[] = [];
  
  // 全要素をトラバースしてID収集
  const collectIds = (element: Element, path: string = '') => {
    const id = element.getAttribute('id');
    if (id) {
      const elementType = element.tagName;
      if (!idMap.has(id)) {
        idMap.set(id, []);
      }
      idMap.get(id)!.push({ elementType, path });
    }
    
    // 子要素を再帰的に処理
    for (let i = 0; i < element.children.length; i++) {
      collectIds(element.children[i], `${path}.${element.tagName}[${i}]`);
    }
  };
  
  collectIds(doc.documentElement);
  
  // 重複チェック
  for (const [id, occurrences] of idMap) {
    if (occurrences.length > 1) {
      const elementTypes = occurrences.map(o => o.elementType).join(', ');
      errors.push(
        `Duplicate ID "${id}" found ${occurrences.length} times in: ${elementTypes}`
      );
    }
  }
  
  return {
    isValid: errors.length === 0,
    errors,
    warnings: [],
    info: [`Total unique IDs: ${idMap.size}`]
  };
}
```
