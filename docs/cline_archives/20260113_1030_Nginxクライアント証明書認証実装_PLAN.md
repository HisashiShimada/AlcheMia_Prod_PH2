## 1. タスク概要

Nginxをリバースプロキシとして使用し、クライアント証明書（mTLS: mutual TLS）による認証を実装します。クライアント証明書から組織ID（organization_id）を自動的に抽出し、バックエンドAPIに渡すことで、ユーザーが手動で組織IDを入力する必要をなくします。

### 5.3 バックエンド実装

#### clientCertMiddleware.ts（新規作成）

```typescript
import { Request, Response, NextFunction } from 'express';
import { getPool } from '../database/db';
import { v4 as uuidv4 } from 'uuid';

/**
 * クライアント証明書ミドルウェア
 * Nginxから渡された証明書情報を検証し、organization_idを抽出する
 */
export async function clientCertMiddleware(
  req: Request,
  res: Response,
  next: NextFunction
): Promise<void> {
  const pool = getPool();
  const requestId = uuidv4();

  try {
    // 1. Nginxから渡された証明書情報を取得
    const certDN = req.headers['x-client-cert-dn'] as string;
    const certCN = req.headers['x-client-cert-cn'] as string;
    const certVerify = req.headers['x-client-cert-verify'] as string;

    // 2. 証明書検証状態をチェック
    if (certVerify !== 'SUCCESS') {
      res.status(403).json({
        success: false,
        error: {
          code: 'CERT_VERIFICATION_FAILED',
          message: 'クライアント証明書の検証に失敗しました',
        },
        meta: {
          timestamp: new Date().toISOString(),
          requestId,
        },
      });
      return;
    }

    // 3. CNから組織IDを抽出
    if (!certCN) {
      res.status(403).json({
        success: false,
        error: {
          code: 'CERT_CN_MISSING',
          message: 'クライアント証明書にCNが含まれていません',
        },
        meta: {
          timestamp: new Date().toISOString(),
          requestId,
        },
      });
      return;
    }

    const organizationId = certCN.trim();

    // 4. 組織IDの形式チェック
    if (!organizationId.startsWith('org_')) {
      res.status(403).json({
        success: false,
        error: {
          code: 'CERT_INVALID_ORG_ID',
          message: '証明書のCNが無効な形式です',
        },
        meta: {
          timestamp: new Date().toISOString(),
          requestId,
        },
      });
      return;
    }

    // 5. データベースで組織の存在確認
    const orgQuery = `
      SELECT organization_id, organization_name, is_active
      FROM organization
      WHERE organization_id = $1
    `;
    
    const orgResult = await pool.query(orgQuery, [organizationId]);

    if (orgResult.rows.length === 0) {
      res.status(403).json({
        success: false,
        error: {
          code: 'CERT_ORG_NOT_FOUND',
          message: '証明書の組織が見つかりません',
        },
        meta: {
          timestamp: new Date().toISOString(),
          requestId,
        },
      });
      return;
    }

    const org = orgResult.rows[0];

    // 6. 組織が有効かチェック
    if (!org.is_active) {
      res.status(403).json({
        success: false,
        error: {
          code: 'CERT_ORG_INACTIVE',
          message: '組織が無効化されています',
        },
        meta: {
          timestamp: new Date().toISOString(),
          requestId,
        },
      });
      return;
    }

    // 7. organization_idをX-Organization-Idヘッダーとして設定
    // これにより既存のtenantIsolationミドルウェアと統合
    req.headers['x-organization-id'] = organizationId;

    // 8. 証明書情報をreqに保存（オプション）
    (req as any).clientCert = {
      dn: certDN,
      cn: certCN,
      organizationId,
      organizationName: org.organization_name,
    };

    next();
  } catch (error) {
    console.error('Client certificate middleware error:', error);
    res.status(500).json({
      success: false,
      error: {
        code: 'SYSTEM_ERROR',
        message: 'システムエラーが発生しました',
      },
      meta: {
        timestamp: new Date().toISOString(),
        requestId,
      },
    });
  }
}
```

### 5.5 証明書生成スクリプト

```bash
#!/bin/bash
# certs/generate-certs.sh

set -e

CERTS_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
CA_DIR="$CERTS_DIR/ca"
CLIENTS_DIR="$CERTS_DIR/clients"

# ディレクトリ作成
mkdir -p "$CA_DIR"
mkdir -p "$CLIENTS_DIR"

echo "========================================="
echo "1. CA証明書の生成"
echo "========================================="

# CA秘密鍵の生成
openssl genrsa -out "$CA_DIR/ca.key" 4096

# CA証明書の生成（10年有効）
openssl req -new -x509 -days 3650 -key "$CA_DIR/ca.key" -out "$CA_DIR/ca.crt" \
  -subj "/C=JP/ST=Tokyo/L=Tokyo/O=MMDPF/OU=Certificate Authority/CN=MMDPF Root CA"

echo "CA証明書を生成しました: $CA_DIR/ca.crt"

# 組織IDリスト
ORGANIZATIONS=(
  "org_maiml_lab_001"
  "org_maiml_lab_002"
)

for ORG_ID in "${ORGANIZATIONS[@]}"; do
  echo ""
  echo "========================================="
  echo "2. クライアント証明書の生成: $ORG_ID"
  echo "========================================="
  
  # クライアント秘密鍵の生成
  openssl genrsa -out "$CLIENTS_DIR/${ORG_ID}.key" 2048
  
  # CSR（証明書署名要求）の生成
  openssl req -new -key "$CLIENTS_DIR/${ORG_ID}.key" -out "$CLIENTS_DIR/${ORG_ID}.csr" \
    -subj "/C=JP/ST=Tokyo/L=Tokyo/O=MMDPF/OU=${ORG_ID}/CN=${ORG_ID}"
  
  # クライアント証明書の生成（CA署名、3年有効）
  openssl x509 -req -days 1095 \
    -in "$CLIENTS_DIR/${ORG_ID}.csr" \
    -CA "$CA_DIR/ca.crt" \
    -CAkey "$CA_DIR/ca.key" \
    -CAcreateserial \
    -out "$CLIENTS_DIR/${ORG_ID}.crt"
  
  # PKCS#12形式に変換（ブラウザインポート用）
  openssl pkcs12 -export \
    -in "$CLIENTS_DIR/${ORG_ID}.crt" \
    -inkey "$CLIENTS_DIR/${ORG_ID}.key" \
    -out "$CLIENTS_DIR/${ORG_ID}.p12" \
    -name "$ORG_ID" \
    -passout pass:mmdpf123
  
  # CSRファイルを削除
  rm "$CLIENTS_DIR/${ORG_ID}.csr"
  
  echo "クライアント証明書を生成しました:"
  echo "  - 証明書: $CLIENTS_DIR/${ORG_ID}.crt"
  echo "  - 秘密鍵: $CLIENTS_DIR/${ORG_ID}.key"
  echo "  - PKCS#12: $CLIENTS_DIR/${ORG_ID}.p12 (パスワード: mmdpf123)"
done

echo ""
echo "========================================="
echo "証明書生成完了"
echo "========================================="
echo "CA証明書: $CA_DIR/ca.crt"
echo "クライアント証明書: $CLIENTS_DIR/"
echo ""
echo "次のステップ:"
echo "1. Nginxに ca.crt を配置"
echo "2. クライアントに .p12 ファイルを配布してブラウザにインポート"
```

## 6. セキュリティ考慮事項

### セキュリティ対策

1. ✅ **CA秘密鍵の厳重管理**
   - `certs/ca/ca.key` は `.gitignore` に追加
   - アクセス権限を600に設定
   - 本番環境では別途安全な場所に保管

2. ✅ **証明書の有効期限管理**
   - CA証明書: 10年
   - クライアント証明書: 3年
   - 有効期限切れ前に更新通知

3. ✅ **証明書失効リスト（CRL）**
   - 将来的にCRLまたはOCSPの実装を検討

4. ✅ **証明書とDB組織の照合**
   - CNから抽出した組織IDがDBに存在するか確認
   - 組織が有効（is_active=true）であるか確認

5. ✅ **エラーハンドリング**
   - 証明書なしアクセスを403で拒否
   - 無効な証明書を403で拒否
   - エラーメッセージは最小限（情報漏洩防止）

### リスク評価

| リスク | 影響度 | 発生確率 | 対策 |
|--------|--------|----------|------|
| CA秘密鍵の漏洩 | 高 | 低 | 厳重管理、アクセス制限、定期的な見直し |
| クライアント証明書の不正使用 | 中 | 低 | 失効リスト、有効期限管理 |
| 証明書なしアクセス | 中 | 中 | Nginxでの強制認証、エラーハンドリング |
| 証明書の期限切れ | 中 | 中 | 有効期限監視、自動更新通知 |

## 8. ロールバック計画

問題が発生した場合：

1. **Nginx設定の巻き戻し**
   - `ssl_verify_client` を `optional` に変更（一時的に証明書なしを許可）
   - または `off` に変更（完全無効化）

2. **バックエンドの巻き戻し**
   - `clientCertMiddleware` を無効化
   - 既存の `X-Organization-Id` ヘッダー処理に戻す

3. **フロントエンドの巻き戻し**
   - organizationId入力欄を復活
