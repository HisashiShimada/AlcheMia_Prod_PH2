```typescript
export const uploadFiles = async (params: UploadFilesParams): Promise<UploadResponseData> => {
  const { uploadMode, targetPath, files, metadata, onProgress } = params;
  const client = createApiClient();

  try {
    const formData = new FormData();
    formData.append('uploadMode', uploadMode);
    formData.append('targetPath', targetPath);
    formData.append('filePaths', JSON.stringify(filePaths));
    files.forEach((file) => { formData.append('files', file); });
    if (metadata) {
      formData.append('metadata', JSON.stringify(metadata));
    }

    const response = await client.post<ApiResponse<UploadResponseData>>(
      '/files/upload',
      formData,
      {
        headers: { 'Content-Type': 'multipart/form-data' },
        onUploadProgress: (progressEvent: AxiosProgressEvent) => {
          if (onProgress && progressEvent.total) {
            const percentage = Math.round((progressEvent.loaded * 100) / progressEvent.total);
            onProgress({
              uploadedBytes: progressEvent.loaded,
              totalBytes: progressEvent.total,
              percentage,
              uploadedCount: 0,
              totalCount: files.length,
            });
          }
        },
      }
    );

    if (response.data.success) {
      return response.data.data;
    }
    throw new Error('アップロードに失敗しました');
  } catch (error) {
    console.error('ファイルアップロードエラー:', error);
    throw error;
  }
};
```

```typescript
export async function uploadFiles(req: Request, res: Response): Promise<void> {
  const requestId = uuidv4();
  try {
    const { organizationId, userId } = req.user!;
    const { uploadMode, targetPath, emptyFolders } = req.body;
    const files = req.files as Express.Multer.File[];

    const service = new FileAccessService(repository, accessValidator);
    const result = await service.uploadFiles(
      organizationId, userId, uploadMode, targetPath,
      files, metadata, emptyFolders, filePaths
    );

    res.status(200).json({
      success: true,
      data: result,
      meta: { timestamp: new Date().toISOString(), requestId },
    });
  } catch (error: any) {
    logError('File upload error', error, {
      action: 'file_upload',
      organizationId: req.user?.organizationId,
      requestId,
    });

    let statusCode = 500;
    let errorCode = 'SYSTEM_ERROR';
    let errorMessage = 'システムエラーが発生しました';

    // エラータイプ判別
    if (error.message.includes('パストラバーサル')) {
      statusCode = 400;
      errorCode = 'VALIDATION_ERROR';
    } else if (error.message.includes('アップロード')) {
      statusCode = 500;
      errorCode = 'STORAGE_ERROR';
    }

    res.status(statusCode).json({
      success: false,
      error: { code: errorCode, message: errorMessage },
      meta: { timestamp: new Date().toISOString(), requestId },
    });
  }
}
```

**特徴**:
- ✅ エラーハンドリングあり（エラータイプ判別）

- ✅ 並列処理制御あり（processConcurrently）

## 2. 問題点の分析

### 2.1 アップロードが1度で反映されない原因

| 原因 | 詳細 | 影響度 |
|------|------|--------|
| **ネットワークタイムアウト** | タイムアウト設定がないため、長時間アップロードで切断される | **High** |
| **リトライロジックなし** | 一時的なネットワーク問題で失敗してもリトライしない | **High** |
| **大ファイルでメモリ不足** | メモリストレージ使用で500MB近いファイルでメモリ圧迫 | **Medium** |
| **同時アップロード数制限なし（フロントエンド）** | 100ファイル一斉アップロードでブラウザがハング | **Medium** |
| **エラー情報が不明確** | エラーメッセージが汎用的で原因特定が困難 | **Low** |

### 2.2 技術的制約

| 制約事項 | 現状 | 推奨 |
|---------|------|------|
| Axiosタイムアウト | 未設定（デフォルト: 無制限） | 5分〜10分 |
| S3 PutObjectタイムアウト | 未設定 | 10分 |
| Multerストレージ | メモリ | ディスク推奨（大ファイル対応） |
| マルチパートアップロード | 未対応 | 100MB以上で推奨 |
| 同時アップロード数 | 無制限 | 3〜5ファイル |

### 5.1 S3マルチパートアップロード

**AWS SDK V3の実装例**:
```typescript
import { 
  S3Client,
  CreateMultipartUploadCommand,
  UploadPartCommand,
  CompleteMultipartUploadCommand,
  AbortMultipartUploadCommand,
} from '@aws-sdk/client-s3';

// 1. マルチパートアップロード開始
const createResponse = await s3Client.send(
  new CreateMultipartUploadCommand({
    Bucket: bucketName,
    Key: fileKey,
  })
);
const uploadId = createResponse.UploadId;

// 2. チャンクごとにアップロード
const partSize = 5 * 1024 * 1024; // 5MB
for (let i = 0; i < chunks.length; i++) {
  const uploadPartResponse = await s3Client.send(
    new UploadPartCommand({
      Bucket: bucketName,
      Key: fileKey,
      UploadId: uploadId,
      PartNumber: i + 1,
      Body: chunks[i],
    })
  );
  parts.push({
    PartNumber: i + 1,
    ETag: uploadPartResponse.ETag,
  });
}

// 3. アップロード完了
await s3Client.send(
  new CompleteMultipartUploadCommand({
    Bucket: bucketName,
    Key: fileKey,
    UploadId: uploadId,
    MultipartUpload: { Parts: parts },
  })
);
```

### 5.2 リトライロジック（Exponential Backoff）

```typescript
async function uploadWithRetry(
  uploadFn: () => Promise<void>,
  maxRetries: number = 3
): Promise<void> {
  for (let attempt = 1; attempt <= maxRetries; attempt++) {
    try {
      await uploadFn();
      return; // 成功
    } catch (error) {
      if (attempt === maxRetries) throw error;
      
      const delay = Math.min(1000 * Math.pow(2, attempt), 10000);
      console.log(`Retry ${attempt}/${maxRetries} after ${delay}ms`);
      await new Promise(resolve => setTimeout(resolve, delay));
    }
  }
}
```

### 5.3 同時アップロード数制御

```typescript
async function uploadFilesSequentially(
  files: File[],
  concurrency: number = 3
): Promise<void> {
  const queue = [...files];
  const active = new Set<Promise<void>>();

  while (queue.length > 0 || active.size > 0) {
    while (active.size < concurrency && queue.length > 0) {
      const file = queue.shift()!;
      const promise = uploadSingleFile(file)
        .finally(() => active.delete(promise));
      active.add(promise);
    }
    if (active.size > 0) {
      await Promise.race(active);
    }
  }
}
```
