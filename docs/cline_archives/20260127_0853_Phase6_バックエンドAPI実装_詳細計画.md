import { Request, Response } from 'express';
import { maimlService } from '../services/maimlService';

export const maimlController = {
  // MaiMLファイル解析
  async parse(req: Request, res: Response) {
    try {
      const { content } = req.body;
      const result = await maimlService.parseMaiML(content);
      res.json(result);
    } catch (error) {
      res.status(500).json({ success: false, error: error.message });
    }
  },

  // MaiMLファイル保存
  async save(req: Request, res: Response) {
    try {
      const { projectId, content } = req.body;
      const organizationId = req.organizationId;
      const result = await maimlService.saveMaiML(projectId, content, organizationId);
      res.json(result);
    } catch (error) {
      res.status(500).json({ success: false, error: error.message });
    }
  },

  // MaiMLファイル読込

    try {
      const { filePath } = req.body;
      const organizationId = req.organizationId;

      res.json(result);
    } catch (error) {
      res.status(500).json({ success: false, error: error.message });

      const result = await maimlService.listMaiMLFiles(projectId, organizationId);
      res.json(result);

      res.status(500).json({ success: false, error: error.message });
    }
  },

  async validate(req: Request, res: Response) {
    try {
      const { content } = req.body;
      const result = await maimlService.validateMaiML(content);
      res.json(result);
    } catch (error) {
      res.status(500).json({ success: false, error: error.message });

      res.status(500).json({ success: false, error: error.message });

async parseMaiML(content: string) {
  const parser = new DOMParser();
  const doc = parser.parseFromString(content, 'text/xml');
  
  // maiml要素取得
  const maimlElement = doc.getElementsByTagName('maiml')[0];
  if (!maimlElement) {
    throw new Error('Invalid MAIML format');
  }
  
  // ペトリネット構造抽出
  const places = extractPlaces(doc);
  const transitions = extractTransitions(doc);
  const arcs = extractArcs(doc);
  
  return {
    success: true,
    data: {
      metadata: extractMetadata(doc),
      places,
      transitions,
      arcs,
      statistics: {
        placeCount: places.length,
        transitionCount: transitions.length,
        arcCount: arcs.length
      }
    }
  };
}

/**
 * MaiMLファイル保存
 * Reference版のロジックを統合
 */
async saveMaiML(projectId: string, content: string, organizationId: string) {
  // 1. バリデーション
  const validation = await this.validateMaiML(content);
  if (!validation.isValid) {
    throw new Error(`Validation failed: ${validation.errors.join(', ')}`);
  }
  
  // 2. ファイルパス生成
  const filePath = `org_${organizationId}/projects/${projectId}/${projectId}.maiml`;
  
  // 3. バックアップ作成（既存ファイルがあれば）
  await createBackup(filePath, organizationId);
  
  // 4. S3アップロード（storageService使用）
  await storageService.uploadFile(organizationId, filePath, content, {
    contentType: 'application/xml',
    metadata: {
      'edit-timestamp': new Date().toISOString(),
      'validation-status': 'valid'
    }
  });
  
  // 5. file_metadataテーブル更新
  await updateFileMetadata(filePath, content, organizationId);
  
  return {
    success: true,
    filePath,
    size: Buffer.byteLength(content, 'utf8'),
    lastModified: new Date().toISOString()
  };
}
```

/**
 * MaiMLバリデーション
 * Reference版のロジックを移植
 */
async validateMaiML(content: string) {
  const errors: string[] = [];
  const warnings: string[] = [];
  const info: string[] = [];
  
  try {
    // XML構文チェック
    const parser = new DOMParser({
      errorHandler: {
        warning: (msg: string) => warnings.push(`XML警告: ${msg}`),
        error: (msg: string) => errors.push(`XML構文エラー: ${msg}`),
        fatalError: (msg: string) => errors.push(`XML致命的エラー: ${msg}`)
      }
    });
    
    const doc = parser.parseFromString(content, 'text/xml');
    
    // maiml要素チェック
    const maimlElement = doc.getElementsByTagName('maiml')[0];
    if (!maimlElement) {
      errors.push('maiml要素が見つかりません');
    }
    
    // ノードID重複チェック
    const nodeIds = new Set<string>();
    checkDuplicateIds(doc, nodeIds, errors);
    
    // アーク参照チェック
    checkArcReferences(doc, nodeIds, errors);
    
    const isValid = errors.length === 0;
    
    return {
      success: true,
      isValid,
      errors,
      warnings,
      info
    };
    
  } catch (error) {
    return {
      success: false,
      isValid: false,
      errors: [error.message],
      warnings: [],
      info: []
    };
  }
}
```

async function createBackup(filePath: string, organizationId: string) {
  try {
    // 既存ファイル確認
    const exists = await storageService.fileExists(organizationId, filePath);
    
    if (exists) {
      // 既存ファイルを取得
      const content = await storageService.downloadFile(organizationId, filePath);
      
      // バックアップパス生成
      const timestamp = new Date().toISOString().replace(/[:.]/g, '-');
      const backupPath = filePath.replace('.maiml', `.backup.${timestamp}.maiml`);
      
      // バックアップ保存
      await storageService.uploadFile(organizationId, backupPath, content);
      
      console.log('✅ バックアップ作成:', backupPath);
    }
  } catch (error) {
    console.warn('⚠️ バックアップ作成失敗:', error.message);
    // バックアップ失敗は警告のみ（保存処理は継続）
  }
}
