## 1. タスク概要

現在のロール別セッション有効時間（12〜48時間の固定期限方式）を、全ロール共通の30分スライディング方式に変更し、期限切れ時の自動ログアウト機能を実装します。

### 変更内容サマリー

| 項目 | 現在 | 変更後 |
|------|------|--------|
| **セッション方式** | 固定期限方式（ログイン時から絶対時間） | スライディング方式（最後の操作から相対時間） |
| **有効時間** | ロール別（12〜48時間） | 全ロール共通（30分） |
| **自動ログアウト** | なし（次回API呼び出し時に401エラー） | あり（フロントエンドでタイマー + API Interceptor） |
| **事前警告** | なし | あり（期限5分前にモーダル表示） |

#### jwtService.ts
```typescript
/**
 * JWTトークンをリフレッシュ
 * 既存の有効なトークンから新しいトークンを生成
 */
async refresh(token: string): Promise<Result<string, ServiceError>> {
  try {
    // 既存トークンを検証
    const verifyResult = await this.verify(token);
    if (!verifyResult.success) {
      return verifyResult;
    }

    // 新しいトークンを生成（有効期限を延長）
    const payload = verifyResult.data;
    const newToken = await this.generate({
      userId: payload.userId,
      organizationId: payload.organizationId,
      role: payload.role,
      sessionId: payload.sessionId, // 同じsessionIdを維持
      permissions: payload.permissions,
    });

    return newToken;
  } catch (error) {
    return {
      success: false,
      error: {
        code: 'SYSTEM_JWT_REFRESH_FAILED',
        message: 'Token refresh failed',
        originalError: error as Error
      }
    };
  }
}
```

#### useSessionTimer.ts
```typescript
import { useEffect, useCallback, useRef } from 'react';
import { Modal } from 'antd';
import { useRouter } from 'next/router';

interface UseSessionTimerOptions {
  token: string | null;
  onWarning?: () => void;
  onExpired?: () => void;
}

export function useSessionTimer({ token, onWarning, onExpired }: UseSessionTimerOptions) {
  const router = useRouter();
  const warningTimerRef = useRef<NodeJS.Timeout | null>(null);
  const expiredTimerRef = useRef<NodeJS.Timeout | null>(null);

  const handleWarning = useCallback(() => {
    Modal.warning({
      title: 'セッション期限が近づいています',
      content: 'あと5分でセッションが切れます。作業を保存してください。',
      okText: '了解',
    });
    onWarning?.();
  }, [onWarning]);

  const handleExpired = useCallback(() => {
    // sessionStorageクリア
    sessionStorage.clear();
    
    // ログイン画面へリダイレクト
    router.push('/login');
    
    // メッセージ表示
    Modal.info({
      title: 'セッションが期限切れになりました',
      content: '再度ログインしてください。',
    });
    
    onExpired?.();
  }, [router, onExpired]);

  useEffect(() => {
    if (!token) {
      return;
    }

    try {
      // JWTをデコード
      const payload = JSON.parse(atob(token.split('.')[1]));
      const expiresAt = payload.exp * 1000;
      const now = Date.now();
      const timeUntilExpiry = expiresAt - now;

      if (timeUntilExpiry <= 0) {
        // 既に期限切れ
        handleExpired();
        return;
      }

      // 期限の5分前に警告を表示
      const warningTime = timeUntilExpiry - 5 * 60 * 1000;
      if (warningTime > 0) {
        warningTimerRef.current = setTimeout(handleWarning, warningTime);
      }

      // 期限時に自動ログアウト
      expiredTimerRef.current = setTimeout(handleExpired, timeUntilExpiry);
    } catch (error) {
      console.error('Failed to decode JWT:', error);
    }

    // クリーンアップ
    return () => {
      if (warningTimerRef.current) {
        clearTimeout(warningTimerRef.current);
      }
      if (expiredTimerRef.current) {
        clearTimeout(expiredTimerRef.current);
      }
    };
  }, [token, handleWarning, handleExpired]);
}
```

#### axiosConfig.ts
```typescript
import axios from 'axios';

// レスポンスインターセプター
axios.interceptors.response.use(
  (response) => {
    // レスポンスヘッダーから新しいトークンを取得
    const newToken = response.headers['x-new-token'];
    
    if (newToken) {
      // sessionStorageのトークンを更新
      sessionStorage.setItem('token', newToken);
      
      // axiosのデフォルトヘッダーも更新
      axios.defaults.headers.common['Authorization'] = `Bearer ${newToken}`;
    }
    
    return response;
  },
  (error) => {
    // 401エラー（トークン期限切れ）時の処理
    if (error.response?.status === 401 && 
        error.response?.data?.error?.code === 'AUTH_TOKEN_EXPIRED') {
      // sessionStorageクリア
      sessionStorage.clear();
      
      // ログイン画面へリダイレクト
      window.location.href = '/login';
      
      // エラーメッセージ
      Modal.error({
        title: 'セッションが期限切れになりました',
        content: '再度ログインしてください。',
      });
    }
    
    return Promise.reject(error);
  }
);
