export const validateMaimlContent = async (
  content: string,
  organizationId: string
): Promise<{
  valid: boolean;
  errors: any[];
  warnings: any[];
}> => {
  try {
    const response = await fetch('/api/v1/maiml/validate', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${localStorage.getItem('token')}`,
      },
      credentials: 'include',
      body: JSON.stringify({
        content,
        organizationId,
      }),
    });
    
    if (!response.ok) {
      throw new Error('バリデーションAPIエラー');
    }
    
    const result = await response.json();
    return result.data;
  } catch (error) {
    console.error('❌ [apiClient] バリデーションエラー:', error);
    throw error;
  }
};
