#### 条件分岐

```typescript
<Modal title={editingOrg ? '組織編集' : '組織追加'} ...>
  <Form form={form} layout="vertical">
    {editingOrg ? (
      // 編集モーダル: 現在のレイアウト
      <>...</>
    ) : (
      // 追加モーダル: 新しいレイアウト
      <>
        <Divider orientation="left">組織情報</Divider>
        {/* 組織情報フィールド群 */}
        
        <Divider orientation="left">代表者情報</Divider>
        {/* 代表者情報フィールド群 */}
      </>
    )}
  </Form>
</Modal>
```

#### handleModalOk関数の修正
```typescript
const handleModalOk = async () => {
  try {
    const values = await form.validateFields();
    
    // 代表者メールアドレスをフォームデータから除外（UI表示のみのため）
    const { chiefUserEmail, ...saveData } = values;
    
    const client = createApiClient();
    
    if (editingOrg) {
      await client.put(`/provider/organization/${editingOrg.organizationId}`, saveData);
      messageApi.success('組織情報を更新しました');
    } else {
      await client.post('/provider/organization', saveData);
      messageApi.success('組織を追加しました');
    }
    
    setIsModalVisible(false);
    form.resetFields();
    await fetchOrganizations();
  } catch (error) {
    // エラーハンドリング
  }
};
```
