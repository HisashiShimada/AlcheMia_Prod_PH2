#### 状態管理
```typescript
const [isLoading, setIsLoading] = useState(false);
const [isSaving, setIsSaving] = useState(false);
const [error, setError] = useState<string | null>(null);
const [maimlContent, setMaimlContent] = useState<string>('');
```

#### ファイル読み込み
```typescript
useEffect(() => {
  if (open && fileData) {
    setIsLoading(true);
    setError(null);
    
    try {
      // ファイル読み込み
      const content = fileData.content;
      setMaimlContent(content);
    } catch (err) {
      setError('ファイルの読み込みに失敗しました');
      console.error(err);
    } finally {
      setIsLoading(false);
    }
  }
}, [open, fileData]);
```

#### 保存処理
```typescript
const handleSave = useCallback(async (content: string) => {
  if (!fileData || !onSave) return;
  
  setIsSaving(true);
  setError(null);
  
  try {
    await onSave(fileData.filename, content);
    message.success('保存しました');
  } catch (err) {
    setError('保存に失敗しました');
    message.error('保存に失敗しました');
    console.error(err);
  } finally {
    setIsSaving(false);
  }
}, [fileData, onSave]);
```

try {

  message.success('保存しました');
} catch (err: any) {
  if (err.response?.status === 401) {
    message.error('認証エラー: ログインし直してください');
  } else if (err.response?.status === 403) {
    message.error('権限エラー: この操作は許可されていません');

    message.error('ファイルが見つかりません');

    message.error('保存に失敗しました');
  }

}
