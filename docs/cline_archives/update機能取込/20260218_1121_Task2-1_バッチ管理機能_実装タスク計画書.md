## 📖 バッチ管理機能の説明

### 1. バッチ管理とは

**バッチ管理**は、複数のファイルアップロードを1つの「バッチ」としてグループ化し、全体の進捗状況やエラー情報を追跡する仕組みです。

### 2. Reference版のバッチフロー

```javascript
// Step 1: バッチID生成
const batchId = utils.generateBatchId(); // 例: "batch_20260218_112100_abc123"

// Step 2: バッチ作成（DB登録）
await dbOperations.createUploadBatch({
  batch_id: batchId,
  batch_name: batchName || `Upload ${new Date().toISOString()}`,
  uploaded_by: userId,
  organization_id: organizationId,
  upload_type: 'single' | 'bulk' | 'folder',  // アップロード種別
  total_files: files.length,                   // 合計ファイル数
  status: 'in_progress'                        // 初期状態
});

// Step 3: 各ファイルアップロード時にバッチIDを紐付け
for (const file of files) {
  const fileMetadata = {
    file_id: generateFileId(),
    file_name: file.originalname,
    upload_batch_id: batchId,  // ← バッチIDを紐付け
    // ... その他のメタデータ
  };
  
  await dbOperations.saveFileMetadata(fileMetadata);
}

// Step 4: バッチ完了（進捗状況を更新）
await dbOperations.updateUploadBatch(batchId, {
  completed_files: successCount,
  failed_files: failCount,
  status: successCount === files.length ? 'completed' : 'completed_with_errors',
  end_time: new Date()
});
```

### 3. バッチIDの形式

```
batch_[YYYYMMDD]_[HHMMSS]_[ランダム文字列]

例:
- batch_20260218_112100_a1b2c3
- batch_20260218_153045_x9y8z7
```

### 4. バッチのステータス

| ステータス | 説明 | 遷移条件 |
|-----------|------|---------|
| `in_progress` | アップロード中 | バッチ作成直後 |
| `completed` | 完了（全成功） | 全ファイルが成功 |
| `completed_with_errors` | 完了（一部失敗） | 一部ファイルが失敗 |
| `failed` | 失敗（全失敗） | 全ファイルが失敗 |

### 5. バッチ管理のメリット

#### ① アップロード履歴の追跡
```sql
-- 特定ユーザーのアップロード履歴を取得
SELECT * FROM upload_batch 
WHERE uploaded_by = 'user_id' 
ORDER BY created_at DESC;
```

#### ② 進捗状況の監視
```sql
-- 進行中のバッチ一覧
SELECT 
  batch_id,
  batch_name,
  completed_files,
  total_files,
  ROUND(completed_files::NUMERIC / total_files * 100, 2) as progress_percent
FROM upload_batch
WHERE status = 'in_progress';
```

#### ③ エラー分析
```sql
-- エラーが発生したファイルを特定
SELECT fm.* 
FROM file_metadata fm
JOIN upload_batch ub ON fm.upload_batch_id = ub.batch_id
WHERE ub.batch_id = 'batch_20260218_112100_abc123'
  AND fm.upload_status = 'failed';
```

#### ④ 一括削除・ロールバック
```sql
-- バッチ単位でファイルを削除
DELETE FROM file_metadata 
WHERE upload_batch_id = 'batch_20260218_112100_abc123';
```

## 🗄️ DBスキーマ設計

### 1. upload_batchテーブル

```sql
CREATE TABLE upload_batch (
  -- 主キー
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  
  -- バッチ識別子
  batch_id VARCHAR(100) UNIQUE NOT NULL,
  
  -- バッチ情報
  batch_name VARCHAR(255) NOT NULL,
  upload_type VARCHAR(50) NOT NULL,  -- 'single', 'bulk', 'folder'
  
  -- 組織・ユーザー
  organization_id UUID NOT NULL REFERENCES organizations(id),
  uploaded_by UUID NOT NULL REFERENCES users(id),
  
  -- 進捗情報
  total_files INTEGER NOT NULL,
  completed_files INTEGER DEFAULT 0,
  failed_files INTEGER DEFAULT 0,
  
  -- ステータス
  status VARCHAR(50) NOT NULL DEFAULT 'in_progress',
  -- 'in_progress', 'completed', 'completed_with_errors', 'failed'
  
  -- タイムスタンプ
  start_time TIMESTAMP NOT NULL DEFAULT NOW(),
  end_time TIMESTAMP,
  created_at TIMESTAMP NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMP NOT NULL DEFAULT NOW(),
  
  -- インデックス
  CONSTRAINT check_completed_files CHECK (completed_files >= 0),
  CONSTRAINT check_failed_files CHECK (failed_files >= 0),
  CONSTRAINT check_total_files CHECK (total_files > 0),
  CONSTRAINT check_progress CHECK (completed_files + failed_files <= total_files)
);

-- インデックス作成
CREATE INDEX idx_upload_batch_organization ON upload_batch(organization_id);
CREATE INDEX idx_upload_batch_uploaded_by ON upload_batch(uploaded_by);
CREATE INDEX idx_upload_batch_status ON upload_batch(status);
CREATE INDEX idx_upload_batch_created_at ON upload_batch(created_at DESC);
```

### 2. file_metadataテーブルの拡張

ALTER TABLE file_metadata

### Phase 2-1-3: Repository層の実装（3日間）

#### backend/src/repositories/UploadBatchRepository.ts

```typescript
import db from '../database/db';
import { 
  UploadBatch, 
  CreateBatchParams, 
  UpdateBatchParams,
  BatchSearchParams,
  BatchProgress 
} from '../types/uploadBatch';

export class UploadBatchRepository {
  /**
   * バッチID生成
   */
  generateBatchId(): string {
    const now = new Date();
    const dateStr = now.toISOString().slice(0, 10).replace(/-/g, '');
    const timeStr = now.toISOString().slice(11, 19).replace(/:/g, '');
    const random = Math.random().toString(36).substring(2, 8);
    return `batch_${dateStr}_${timeStr}_${random}`;
  }

  /**
   * バッチ作成
   */
  async create(params: CreateBatchParams): Promise<UploadBatch> {
    const batchId = this.generateBatchId();
    const batchName = params.batchName || `Upload ${new Date().toISOString()}`;

    const query = `
      INSERT INTO upload_batch (
        batch_id, batch_name, upload_type, organization_id, 
        uploaded_by, total_files, status
      )
      VALUES ($1, $2, $3, $4, $5, $6, 'in_progress')
      RETURNING *
    `;

    const values = [
      batchId,
      batchName,
      params.uploadType,
      params.organizationId,
      params.uploadedBy,
      params.totalFiles
    ];

    const result = await db.query(query, values);
    return this.mapRowToBatch(result.rows[0]);
  }

  /**
   * バッチ更新
   */
  async update(batchId: string, params: UpdateBatchParams): Promise<UploadBatch> {
    const setClauses: string[] = [];
    const values: any[] = [];
    let paramIndex = 1;

    if (params.completedFiles !== undefined) {
      setClauses.push(`completed_files = $${paramIndex++}`);
      values.push(params.completedFiles);
    }

    if (params.failedFiles !== undefined) {
      setClauses.push(`failed_files = $${paramIndex++}`);
      values.push(params.failedFiles);
    }

    if (params.status) {
      setClauses.push(`status = $${paramIndex++}`);
      values.push(params.status);
    }

    if (params.endTime) {
      setClauses.push(`end_time = $${paramIndex++}`);
      values.push(params.endTime);
    }

    setClauses.push(`updated_at = NOW()`);
    values.push(batchId);

    const query = `
      UPDATE upload_batch
      SET ${setClauses.join(', ')}
      WHERE batch_id = $${paramIndex}
      RETURNING *
    `;

    const result = await db.query(query, values);
    if (result.rows.length === 0) {
      throw new Error(`Batch not found: ${batchId}`);
    }

    return this.mapRowToBatch(result.rows[0]);
  }

  /**
   * バッチ取得（ID指定）
   */
  async findById(batchId: string): Promise<UploadBatch | null> {
    const query = 'SELECT * FROM upload_batch WHERE batch_id = $1';
    const result = await db.query(query, [batchId]);
    
    if (result.rows.length === 0) {
      return null;
    }

    return this.mapRowToBatch(result.rows[0]);
  }

  /**
   * バッチ検索
   */
  async search(params: BatchSearchParams): Promise<UploadBatch[]> {
    const conditions: string[] = ['organization_id = $1'];
    const values: any[] = [params.organizationId];
    let paramIndex = 2;

    if (params.uploadedBy) {
      conditions.push(`uploaded_by = $${paramIndex++}`);
      values.push(params.uploadedBy);
    }

    if (params.status) {
      conditions.push(`status = $${paramIndex++}`);
      values.push(params.status);
    }

    if (params.startDate) {
      conditions.push(`created_at >= $${paramIndex++}`);
      values.push(params.startDate);
    }

    if (params.endDate) {
      conditions.push(`created_at <= $${paramIndex++}`);
      values.push(params.endDate);
    }

    const limit = params.limit || 50;
    const offset = params.offset || 0;

    const query = `
      SELECT * FROM upload_batch
      WHERE ${conditions.join(' AND ')}
      ORDER BY created_at DESC
      LIMIT $${paramIndex++} OFFSET $${paramIndex}
    `;

    values.push(limit, offset);

    const result = await db.query(query, values);
    return result.rows.map(row => this.mapRowToBatch(row));
  }

  /**
   * バッチ進捗取得
   */
  async getProgress(batchId: string): Promise<BatchProgress | null> {
    const query = `
      SELECT 
        batch_id,
        batch_name,
        total_files,
        completed_files,
        failed_files,
        status,
        ROUND(completed_files::NUMERIC / total_files * 100, 2) as progress_percent
      FROM upload_batch
      WHERE batch_id = $1
    `;

    const result = await db.query(query, [batchId]);
    
    if (result.rows.length === 0) {
      return null;
    }

    const row = result.rows[0];
    return {
      batchId: row.batch_id,
      batchName: row.batch_name,
      totalFiles: row.total_files,
      completedFiles: row.completed_files,
      failedFiles: row.failed_files,
      progressPercent: parseFloat(row.progress_percent),
      status: row.status
    };
  }

  /**
   * バッチ削除
   */
  async delete(batchId: string): Promise<void> {
    const query = 'DELETE FROM upload_batch WHERE batch_id = $1';
    await db.query(query, [batchId]);
  }

  /**
   * DB行をUploadBatch型にマッピング
   */
  private mapRowToBatch(row: any): UploadBatch {
    return {
      id: row.id,
      batchId: row.batch_id,
      batchName: row.batch_name,
      uploadType: row.upload_type,
      organizationId: row.organization_id,
      uploadedBy: row.uploaded_by,
      totalFiles: row.total_files,
      completedFiles: row.completed_files,
      failedFiles: row.failed_files,
      status: row.status,
      startTime: row.start_time,
      endTime: row.end_time,
      createdAt: row.created_at,
      updatedAt: row.updated_at
    };
  }
}

export default new UploadBatchRepository();
```

/**
 * バッチ履歴取得
 * GET /api/v1/files/batches
 */
router.get(
  '/batches',
  auth,
  tenantIsolation,
  async (req: Request, res: Response) => {
    try {
      const { organizationId } = req as any;
      const { status, startDate, endDate, limit, offset } = req.query;

      const batches = await uploadBatchRepository.search({
        organizationId,
        status: status as any,
        startDate: startDate ? new Date(startDate as string) : undefined,
        endDate: endDate ? new Date(endDate as string) : undefined,
        limit: limit ? parseInt(limit as string) : 50,
        offset: offset ? parseInt(offset as string) : 0
      });

      res.json({
        success: true,
        data: batches,
        meta: {
          timestamp: new Date().toISOString(),
          count: batches.length
        }
      });
    } catch (error) {
      console.error('❌ [Batch] 履歴取得エラー:', error);
      res.status(500).json({
        success: false,
        error: {
          code: 'BATCH_FETCH_ERROR',
          message: 'バッチ履歴の取得に失敗しました'
        }
      });
    }
  }
);

/**
 * バッチ進捗取得
 * GET /api/v1/files/batches/:batchId/progress
 */
router.get(
  '/batches/:batchId/progress',
  auth,
  tenantIsolation,
  async (req: Request, res: Response) => {
    try {
      const { batchId } = req.params;

      const progress = await uploadBatchRepository.getProgress(batchId);

      if (!progress) {
        return res.status(404).json({
          success: false,

        success: true,
        data: progress,
        meta: {
          timestamp: new Date().toISOString()

      });

        error: {
          code: 'BATCH_PROGRESS_ERROR',
