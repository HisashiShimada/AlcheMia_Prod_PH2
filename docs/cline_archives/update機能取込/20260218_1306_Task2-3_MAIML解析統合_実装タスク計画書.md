## 🎯 目的と期待される成果

### 目的

1. **アップロード直後の即時フィードバック**
   - ファイルがMAIMLファイルであるかを即座に判定

   - 適切な役割分担でパフォーマンスを最適化

3. **ファイル妥当性の簡易チェック**
   - MAIMLファイルとして基本的な構造を持っているか
   - 破損ファイルやフォーマット違いを早期検出

## 📖 機能説明

### Reference版の簡易解析実装

```javascript
// Reference版: 簡易MAIML解析
function analyzeMaimlFile(content) {
  try {
    // XMLとして基本的な妥当性チェック
    const isXml = content.includes('<?xml') || content.includes('<maiml') || content.includes('<pnml');
    
    // transition タグの数をカウント
    const transitionMatches = content.match(/<transition/g) || [];
    const transitionsCount = transitionMatches.length;
    
    // place タグの数をカウント
    const placeMatches = content.match(/<place/g) || [];
    const placesCount = placeMatches.length;
    
    return {
      isValid: isXml && (transitionsCount > 0 || placesCount > 0),
      isMaimlFile: content.includes('<maiml') || content.includes('<pnml'),
      transitionsCount,
      placesCount,
      version: null, // 簡易解析ではバージョン情報は取得しない
    };
  } catch (error) {
    return {
      isValid: false,
      isMaimlFile: false,
      transitionsCount: 0,
      placesCount: 0,
      error: error.message,
    };
  }
}
```

### 既存の詳細解析APIとの違い

**ファイル**: `backend/src/utils/maimlAnalyzer.ts`

```typescript
/**
 * MAIML簡易解析ユーティリティ
 * 
 * Phase2-3: MAIML解析統合（2026-02-18）
 * アップロード時の軽量・高速な簡易解析を提供
 * 
 * 詳細解析は既存のmaimlService.parseMAIML()を使用
 */

/**
 * MAIML簡易解析結果
 */
export interface MaimlAnalysisResult {
  /** MAIMLファイルとして妥当か */
  isValid: boolean;
  /** MAIMLファイルであるか（pnmlも含む） */
  isMaimlFile: boolean;
  /** transition要素の数 */
  transitionsCount: number;
  /** place要素の数 */
  placesCount: number;
  /** バージョン情報（簡易解析では取得しない） */
  version: string | null;
  /** エラーメッセージ（解析失敗時） */
  error?: string;
}

/**
 * MAIMLファイルを簡易解析する
 * 
 * @description
 * アップロード時の即時フィードバック用の軽量解析。
 * 正規表現でtransition/placeタグの数をカウントする。
 * 詳細な構造解析は既存のmaimlService.parseMAIML()を使用。
 * 
 * @param content ファイル内容（文字列）
 * @returns MAIML解析結果
 * 
 * @example
 * ```typescript
 * const buffer = file.buffer;
 * const content = buffer.toString('utf-8');
 * const analysis = analyzeMaimlFile(content);
 * 
 * if (analysis.isValid) {
 *   console.log(`Transitions: ${analysis.transitionsCount}`);
 *   console.log(`Places: ${analysis.placesCount}`);
 * }
 * ```
 */
export function analyzeMaimlFile(content: string): MaimlAnalysisResult {
  try {
    // 1. XMLとして基本的な妥当性チェック
    const isXml = 
      content.includes('<?xml') || 
      content.includes('<maiml') || 
      content.includes('<pnml');
    
    // 2. MAIMLファイルかどうかをチェック
    const isMaimlFile = 
      content.includes('<maiml') || 
      content.includes('<pnml');
    
    // 3. transition タグの数をカウント
    const transitionMatches = content.match(/<transition/g);
    const transitionsCount = transitionMatches ? transitionMatches.length : 0;
    
    // 4. place タグの数をカウント
    const placeMatches = content.match(/<place/g);
    const placesCount = placeMatches ? placeMatches.length : 0;
    
    // 5. 妥当性判定
    const isValid = isXml && (transitionsCount > 0 || placesCount > 0);
    
    return {
      isValid,
      isMaimlFile,
      transitionsCount,
      placesCount,
      version: null, // 簡易解析ではバージョン情報は取得しない
    };
  } catch (error) {
    return {
      isValid: false,
      isMaimlFile: false,
      transitionsCount: 0,
      placesCount: 0,
      version: null,
      error: error instanceof Error ? error.message : 'Unknown error',
    };
  }
}

/**
 * ファイルがMAIMLファイルかどうかを判定
 * 
 * @param fileName ファイル名
 * @param contentType Content-Type
 * @returns MAIMLファイルならtrue
 */
export function isMaimlFile(fileName: string, contentType?: string): boolean {
  // 拡張子チェック
  const hasMaimlExtension = fileName.toLowerCase().endsWith('.maiml');
  
  // Content-Typeチェック
  const isXmlContentType = 
    contentType === 'application/xml' || 
    contentType === 'text/xml';
  
  return hasMaimlExtension || isXmlContentType;
}
```

#### Test 1: 単体テスト

**ファイル**: `backend/tests/utils/maimlAnalyzer.test.ts`

```typescript
import { analyzeMaimlFile, isMaimlFile } from '../../src/utils/maimlAnalyzer';

describe('maimlAnalyzer', () => {
  describe('analyzeMaimlFile', () => {
    it('should analyze valid MAIML file', () => {
      const content = `<?xml version="1.0"?>
<maiml>
  <transition id="t1"/>
  <transition id="t2"/>
  <place id="p1"/>
</maiml>`;
      
      const result = analyzeMaimlFile(content);
      
      expect(result.isValid).toBe(true);
      expect(result.isMaimlFile).toBe(true);
      expect(result.transitionsCount).toBe(2);
      expect(result.placesCount).toBe(1);
    });

    it('should detect invalid MAIML file', () => {
      const content = `<?xml version="1.0"?>
<notmaiml>
  <data>test</data>
</notmaiml>`;
      
      const result = analyzeMaimlFile(content);
      
      expect(result.isValid).toBe(false);
      expect(result.isMaimlFile).toBe(false);
    });

    it('should handle empty file', () => {
      const result = analyzeMaimlFile('');
      
      expect(result.isValid).toBe(false);
      expect(result.transitionsCount).toBe(0);
      expect(result.placesCount).toBe(0);
    });

    it('should handle pnml files', () => {
      const content = `<?xml version="1.0"?>
<pnml>
  <transition id="t1"/>
  <place id="p1"/>
</pnml>`;
      
      const result = analyzeMaimlFile(content);
      
      expect(result.isValid).toBe(true);
      expect(result.isMaimlFile).toBe(true);
    });
  });

  describe('isMaimlFile', () => {
    it('should detect maiml extension', () => {
      expect(isMaimlFile('test.maiml')).toBe(true);
      expect(isMaimlFile('Test.MAIML')).toBe(true);
    });

    it('should detect XML content type', () => {
      expect(isMaimlFile('test.xml', 'application/xml')).toBe(true);
      expect(isMaimlFile('test.xml', 'text/xml')).toBe(true);
    });

    it('should return false for non-MAIML files', () => {
      expect(isMaimlFile('test.txt')).toBe(false);
      expect(isMaimlFile('test.json', 'application/json')).toBe(false);
    });
  });
});
```
