// フィールド長制限確認用ユーティリティ関数
function truncateField(value, maxLength, fieldName = 'field') {
  if (!value) return value;
  
  const stringValue = String(value);
  if (stringValue.length <= maxLength) return stringValue;
  
  const truncated = stringValue.substring(0, maxLength);
  console.warn(`[VARCHAR警告] ${fieldName}が${maxLength}文字を超過: "${stringValue}" → "${truncated}"`);
  return truncated;
}

// 使用例

});

/**
 * 文字列バリデーション・サニタイゼーションユーティリティ
 * 
 * Phase3-4: VARCHAR長制限対応（2026-02-18）
 */

/**
 * フィールド値をVARCHAR長制限に合わせて切り詰める
 * 
 * @description
 * DBのVARCHAR制約を超える文字列を自動的に切り詰める。
 * 切り詰め時は末尾に'...'を追加し、警告ログを出力する。
 * 
 * Reference版uploadRoutes.jsのtruncateField()を移植。
 * 
 * @param value 対象の値
 * @param maxLength 最大長
 * @param fieldName フィールド名（ログ用）
 * @returns 切り詰められた文字列（または元の値）
 * 
 * @example
 * ```typescript
 * const fileName = 'very_long_filename_that_exceeds_limit.maiml';
 * const truncated = truncateField(fileName, 50, 'fileName');
 * // 結果: 'very_long_filename_that_exceeds_limi...'
 * ```
 */
export function truncateField(
  value: string | null | undefined,
  maxLength: number,
  fieldName = 'field'
): string | null {
  // nullまたはundefinedの場合はそのまま返す
  if (value === null || value === undefined) {
    return value as null;
  }
  
  // 文字列に変換
  const stringValue = String(value);
  
  // 最大長以下の場合はそのまま返す
  if (stringValue.length <= maxLength) {
    return stringValue;
  }
  
  // 切り詰め（末尾3文字を'...'に置き換え）
  const truncated = stringValue.substring(0, maxLength - 3) + '...';
  
  // 警告ログ出力
  console.warn(
    `⚠️ [Validator] ${fieldName}が${maxLength}文字を超過:`,
    {
      original: stringValue.substring(0, 50) + (stringValue.length > 50 ? '...' : ''),
      originalLength: stringValue.length,
      truncated: truncated,
      truncatedLength: truncated.length,
    }
  );
  
  return truncated;
}

/**
 * 配列の各要素を切り詰める
 * 
 * @param values 対象の配列
 * @param maxLength 各要素の最大長
 * @param fieldName フィールド名（ログ用）
 * @returns 切り詰められた配列
 */
export function truncateArrayFields(
  values: (string | null | undefined)[] | null | undefined,
  maxLength: number,
  fieldName = 'field'
): (string | null)[] | null {
  if (!values) return values as null;
  
  return values.map((value, index) => 
    truncateField(value, maxLength, `${fieldName}[${index}]`)
  );
}

/**
 * file_metadataテーブルの各フィールドの最大長定義
 */
export const FILE_METADATA_MAX_LENGTHS = {
  original_filename: 255,
  file_key: 500,
  file_extension: 10,
  file_category: 50,
  parent_folder_path: 255,
  search_keyword: 50, // 配列の各要素
} as const;
```

import { truncateField, FILE_METADATA_MAX_LENGTHS } from '../utils/stringValidator';

// uploadFilesメソッド内で使用
async uploadFiles(...) {
  // ...

    batchName: batch.batchId,
  });

// saveFileMetadataメソッドも更新
private async saveFileMetadata(data: {...}): Promise<string> {
  const now = new Date();
  
  // Phase3-4: VARCHAR長制限対応
  const fileMetadata = {
    organization_id: data.organizationId,
    file_key: truncateField(data.s3Key, FILE_METADATA_MAX_LENGTHS.file_key, 'file_key'),
    original_filename: truncateField(data.fileName, FILE_METADATA_MAX_LENGTHS.original_filename, 'original_filename'),
    file_size: data.fileSize,
    file_type: data.contentType,
    uploaded_by: data.userId,
    is_maiml_file: data.contentType === 'application/xml' || data.fileName.endsWith('.maiml'),
    s3_bucket: this.storageService.getBucketName(),
    upload_batch_id: data.batchName,
    upload_method: 'web_upload',
    created_at: now,
    updated_at: now,
  };
  
  const result = await this.repository.insertFileMetadata(fileMetadata);
  return result.file_key;
}
