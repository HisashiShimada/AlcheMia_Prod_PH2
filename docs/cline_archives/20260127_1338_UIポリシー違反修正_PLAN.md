## 2. 検出された違反パターン

### 2.1. 色コードのハードコーディング（35箇所）
```typescript
// ❌ 悪い例
style={{ color: '#ff4d4f' }}
style={{ backgroundColor: '#f5f5f5' }}
style={{ borderColor: '#1890ff' }}

// ✅ 良い例
style={{ color: token.colorError }}
style={{ backgroundColor: token.colorBgLayout }}
style={{ borderColor: token.colorPrimary }}
```

### 2.2. フォントサイズの直接指定
```typescript
// ❌ 悪い例
style={{ fontSize: '12px' }}
style={{ fontSize: '11px' }}

// ✅ 良い例
style={{ fontSize: token.fontSizeSM }}
```

### 2.3. スペーシングの具体的な値
```typescript
// ❌ 悪い例
style={{ padding: '16px', margin: '8px' }}

// ✅ 良い例
style={{ padding: token.padding, margin: token.marginXS }}
```

| `8` (padding) | `token.paddingSM` |

| `16` (margin) | `token.margin` |

### 4.2. 修正テンプレート

```typescript
// ファイル冒頭にtheme.useToken()を追加
import { theme } from 'antd';

const ComponentName: React.FC<Props> = (props) => {
  const { token } = theme.useToken();
  
  // ... 既存のコード
  
  return (
    <div style={{
      color: token.colorTextSecondary,    // ❌ '#8c8c8c'
      fontSize: token.fontSizeSM,         // ❌ '12px'
      padding: token.padding,             // ❌ '16px'
      backgroundColor: token.colorBgLayout // ❌ '#f5f5f5'
    }}>
      {/* ... */}
    </div>
  );
};
```
