```typescript
interface PetriNetStructureResponse {
  success: boolean;
  data: {
    fileKey: string;
    // 🆕 複数method対応
```

#### レスポンス構造（TypeScript型定義）

```typescript
interface TransitionStatusResponse {
  success: boolean;
  data: {
    workerId: string;
    maimlFileKey: string;
    transitionStatus: TransitionStatus[];
    summary: {
      totalTransitions: number;
      completed: number;
      inProgress: number;
      paused: number;
      notStarted: number;
      error: number;
      percentage: number;
    };
  };
  timestamp: string;
}

interface TransitionStatus {
  transitionId: string;
  transitionName?: string;
  workId: string;
  latestStatus: string;        // DB値: 'completed', 'in_progress', etc.
  progressStatus: string;      // 日本語: '完了', '作業中', '準備中', '休止中', '未開始', '異常中'
  progressPercentage: number;
  lastUpdated: string;         // ISO 8601 timestamp
  source: string;              // 'worker_progress' or 'default'
}
```

**progressStatusの値一覧:**
- `'完了'` - トランジション完了
- `'作業中'` - 作業進行中
- `'準備中'` - 開始準備中
- `'休止中'` - 一時停止
- `'未開始'` - 未着手
- `'異常中'` - エラー発生

```typescript
/**
 * モックAPIユーティリティ
 * Backend API完成までの暫定実装
 */

import mockPetriNetStructure from '../data/mock/mockPetriNetStructure.json';
import mockTransitionStatus from '../data/mock/mockTransitionStatus.json';
import mockExperimentRecords from '../data/mock/mockExperimentRecords.json';

/**
 * プロジェクトIDをファイルキーから抽出
 */
function extractProjectIdFromFileKey(fileKey: string): string | null {
  const match = fileKey.match(/projects?\/(proj_\d+)/);
  return match ? match[1] : null;
}

/**
 * ワーカーIDとプロジェクトIDからキーを生成
 */
function generateStatusKey(workerId: string, fileKey: string): string {
  const projectId = extractProjectIdFromFileKey(fileKey);
  return `${workerId}_${projectId}`;
}

/**
 * ペトリネット構造取得（モック）
 * @param fileKey - MaiMLファイルキー
 * @returns ペトリネット構造
 */
export async function fetchPetriNetStructureMock(fileKey: string): Promise<any> {
  console.log('🔍 [MockAPI] ペトリネット構造取得（モック）:', fileKey);
  
  // 遅延をシミュレート（300ms）
  await new Promise(resolve => setTimeout(resolve, 300));
  
  const projectId = extractProjectIdFromFileKey(fileKey);
  if (!projectId) {
    throw new Error('Invalid file key format');
  }
  
  const data = (mockPetriNetStructure as any)[projectId];
  if (!data) {
    throw new Error(`No mock data found for project: ${projectId}`);
  }
  
  console.log('✅ [MockAPI] ペトリネット構造取得成功（モック）');
  return data;
}

/**
 * トランジション進捗取得（モック）
 * @param workerId - ワーカーID
 * @param fileKey - MaiMLファイルキー
 * @returns トランジション進捗
 */
export async function fetchTransitionStatusMock(
  workerId: string,
  fileKey: string
): Promise<any> {
  console.log('📊 [MockAPI] トランジション進捗取得（モック）:', { workerId, fileKey });
  
  // 遅延をシミュレート（200ms）
  await new Promise(resolve => setTimeout(resolve, 200));
  
  const statusKey = generateStatusKey(workerId, fileKey);
  const data = (mockTransitionStatus as any)[statusKey];
  
  if (!data) {
    // データがない場合はデフォルト（空の進捗）を返す
    console.log('⚠️ [MockAPI] トランジション進捗データなし、デフォルト返却');
    return {
      success: true,
      data: {
        workerId,
        maimlFileKey: fileKey,
        transitionStatus: [],
        summary: {
          totalTransitions: 0,
          completed: 0,
          inProgress: 0,
          paused: 0,
          notStarted: 0,
          error: 0,
          percentage: 0
        }
      },
      timestamp: new Date().toISOString()
    };
  }
  
  console.log('✅ [MockAPI] トランジション進捗取得成功（モック）');
  return data;
}

/**
 * 実験記録取得（モック）
 * @param workId - 作業ID
 * @returns 実験記録
 */
export async function fetchExperimentRecordsMock(workId: string): Promise<any> {
  console.log('🧪 [MockAPI] 実験記録取得（モック）:', workId);
  
  // 遅延をシミュレート（250ms）
  await new Promise(resolve => setTimeout(resolve, 250));
  
  const data = (mockExperimentRecords as any)[workId];
  
  if (!data) {
    // データがない場合は空のレコードを返す
    console.log('⚠️ [MockAPI] 実験記録データなし、空レコード返却');
    return {
      success: true,
      workId,
      data: {
        materials: [],
        conditions: [],
        results: []
      },
      totalRecords: 0
    };
  }
  
  console.log('✅ [MockAPI] 実験記録取得成功（モック）');
  return data;
}

/**
 * モックAPIかどうかを判定
 * 環境変数で切り替え可能
 */
export function shouldUseMockAPI(): boolean {
  // 開発環境ではモックを使用、本番環境では実APIを使用
  return process.env.REACT_APP_USE_MOCK_API === 'true' || 
         process.env.NODE_ENV === 'development';
}
```

```typescript
// 実験記録データ取得（workId必須）
let experimentRecordsData = null;
if (workId) {
```

## 8. まとめ

### 8.1. 原因の特定

進捗確認画面の詳細ダイアログで実験フロー図が表示されない根本原因は、**PetriNetVisualizationコンポーネントが3つの未実装APIに依存している**ためです。

**未実装API:**
1. `GET /api/worker/petri-net-structure/:fileKey`
2. `GET /api/worker/transition-status/:workerId/:maimlFileKey`
3. `GET /api/experiment-records/by-work-id/:workId`

### 8.2. 推奨される解決策

**既存のPetriNetVisualizationを踏襲し、APIレスポンスをモックデータで代替する方針**

**主なメリット:**
- ✅ 既存コードを最大限活用
- ✅ Backend API完成後は処理を置き換えるだけ
- ✅ reference/backend実装に準拠した型定義
- ✅ 短期間で修正可能（5-7.5時間）
- ✅ 環境変数で簡単に切り替え

### 8.3. 実装のポイント

1. **reference/backend実装を参考にする**: 正確な型定義とレスポンス構造
2. **環境変数での切り替え**: `REACT_APP_USE_MOCK_API`で開発/本番を切り替え
3. **段階的な実装**: ペトリネット構造 → 進捗 → 実験記録の順
4. **詳細なログ**: デバッグを容易にするため

### 8.4. 次のステップ

このレポートに基づき、以下の順序で実装を進めることを推奨します：

1. ✅ **このレポートをユーザーに提示**: 方針の承認を得る
2. 📋 **Phase 1実装開始**: モックデータファイル作成
3. 🔧 **Phase 2-3実装**: ユーティリティとコンポーネント修正
4. ⚙️ **Phase 4-5実装**: 環境変数設定とテスト
5. 🚀 **Backend完成後**: 実API移行

## 付録B: トラブルシューティング

### B.1. モックデータが読み込まれない

**症状**: コンソールに "No mock data found for project" エラー

**原因**: プロジェクトIDの不一致

**対策**:
1. `mockPetriNetStructure.json`のキーを確認
2. ファイルキーからプロジェクトIDが正しく抽出されているか確認
3. `extractProjectIdFromFileKey`関数の正規表現を確認

### B.2. 実験フロー図が空白

**症状**: ダイアログは開くが、フロー図が表示されない

**原因**: ペトリネット構造データの形式不正

**対策**:
1. `places`, `transitions`, `arcs`配列が正しく定義されているか確認
2. ブラウザコンソールでデータ構造をログ出力
3. ReactFlowのエラーメッセージを確認

### B.3. 環境変数が反映されない

**症状**: `REACT_APP_USE_MOCK_API=false`でもモックが使われる

**原因**: 環境変数の読み込みタイミング

**対策**:
1. `.env.development`ファイルが正しい位置にあるか確認
2. フロントエンドを再起動（`npm run dev`）
3. ブラウザのキャッシュをクリア
