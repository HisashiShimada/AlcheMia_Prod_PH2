```typescript
setNodes(nds => nds.map(node => ({
```

```typescript
setSavedPetriNetViewport(currentViewport);

setTimeout(() => {
  setViewport(savedPetriNetViewport, { duration: 300 });
}, 100);
```

#### 3. useMemoによるパフォーマンス最適化
```typescript
// Before (useEffect)
useEffect(() => {
  const templates = extractTemplates(nodes);
  setCategorizedTemplates(templates);
}, [nodes]);

// After (useMemo)
const categorizedTemplates = useMemo(() => 
  extractTemplates(nodes),
  [nodes, extractTemplates]
);
```
