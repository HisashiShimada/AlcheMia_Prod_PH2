### 主要な変更点

#### 1. メソッド一覧の取得
```typescript
// プロジェクトデータからメソッド一覧を取得
const methods = useMemo<Method[]>(() => {
  return projectData?.methods || [];
}, [projectData]);
```

#### 2. 選択中のメソッドIDの管理
```typescript
// 選択中のメソッドID（タブのactiveKey）
const [activeMethodId, setActiveMethodId] = useState<string | null>(null);

// 初期化時に最初のメソッドを選択
useEffect(() => {
  if (open && methods.length > 0 && !activeMethodId) {
    setActiveMethodId(methods[0].methodId);
  }
}, [open, methods, activeMethodId]);
```

#### 3. タスクのフィルタリング
```typescript
// 選択されたメソッドに紐づくタスクのみを取得
const filteredTasksByMethod = useMemo(() => {
  if (!activeMethodId) return [];
  return allTasks.filter(task => task.methodId === activeMethodId);
}, [allTasks, activeMethodId]);

// さらに検索条件でフィルタリング
const filteredTasks = useMemo(() => {
  return filteredTasksByMethod.filter(task => {
    // 検索条件の適用
  });
}, [filteredTasksByMethod, searchText, selectedTaskName, selectedStatus]);
```

#### 4. タブUIの実装
```typescript
const methodTabs = useMemo(() => {
  return methods.map((method) => ({
    key: method.methodId,
    label: method.methodName,
    children: (
      <Table
        columns={columns}
        dataSource={filteredTasks}
        // ... その他のプロパティ
      />
    ),
  }));
}, [methods, filteredTasks]);
```
