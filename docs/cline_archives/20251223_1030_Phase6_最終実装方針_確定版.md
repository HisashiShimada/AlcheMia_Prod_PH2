```typescript
      <PetriNetEditorScreen

  // 通常の管理画面（既存のレンダリング）
```

```typescript
// frontend/src/components/viewer/PetriNetEditorScreen.tsx

'use client';

import React, { useState, useEffect, useCallback } from 'react';
import { Button, Spin, message } from 'antd';
import { ArrowLeftOutlined } from '@ant-design/icons';
import PetriNetEditor from '@/components/viewer/PetriNetEditor/PetriNetEditor';

interface PetriNetEditorScreenProps {
  projectId: string;
  onClose: () => void;
}

/**
 * PetriNetエディタのフルスクリーン表示画面
 * 
 * ExperimentAdminMenuから呼び出される
 */
export default function PetriNetEditorScreen({
  projectId,
  onClose
}: PetriNetEditorScreenProps) {
  const [isLoading, setIsLoading] = useState(true);
  const [projectData, setProjectData] = useState<any>(null);
  const [maimlContent, setMaimlContent] = useState<string | undefined>(undefined);
  const [hasUnsavedChanges, setHasUnsavedChanges] = useState(false);

  // プロジェクト情報とファイルを読み込む
  useEffect(() => {
    loadProjectData();
  }, [projectId]);

  const loadProjectData = async () => {
    try {
      setIsLoading(true);
      
      // プロジェクト情報を取得（仮実装 - 後でAPI実装）
      const project = {
        id: projectId,
        name: `プロジェクト ${projectId}`
      };
      setProjectData(project);
      
      // MAIMLファイルを検索して読み込み
      // TODO: API実装
      
      setMaimlContent(undefined); // 新規作成
    } catch (error) {
      console.error('Failed to load project data:', error);
      message.error('プロジェクトの読み込みに失敗しました');
    } finally {
      setIsLoading(false);
    }
  };

  // 戻るボタン
  const handleBack = useCallback(() => {
    if (hasUnsavedChanges) {
      if (window.confirm('保存されていない変更があります。破棄して戻りますか？')) {
        onClose();
      }
    } else {
      onClose();
    }
  }, [hasUnsavedChanges, onClose]);

  // 保存処理
  const handleSave = useCallback(async (content: string) => {
    try {
      // TODO: 保存API実装
      message.success('保存しました');
      setHasUnsavedChanges(false);
    } catch (error) {
      message.error('保存に失敗しました');
      throw error;
    }
  }, [projectId]);

  if (isLoading) {
    return (
      <div style={{ 
        display: 'flex', 
        justifyContent: 'center', 
        alignItems: 'center', 
        height: '100vh' 
      }}>
        <Spin size="large" tip="読み込み中..." />
      </div>
    );
  }

  return (
    <div style={{ 
      width: '100vw', 
      height: '100vh', 
      display: 'flex', 
      flexDirection: 'column',
      overflow: 'hidden'
    }}>
      {/* ヘッダー */}
      <div style={{ 
        padding: '16px', 
        borderBottom: '1px solid #f0f0f0',
        display: 'flex',
        alignItems: 'center',
        gap: '16px',
        backgroundColor: 'white'
      }}>
        <Button 
          icon={<ArrowLeftOutlined />} 
          onClick={handleBack}
        >
          戻る
        </Button>
        <h1 style={{ margin: 0, fontSize: '18px', fontWeight: 600 }}>
          実験フロー編集 - {projectData?.name}
        </h1>
      </div>

      {/* エディタ本体 */}
      <div style={{ flex: 1, overflow: 'hidden' }}>
        <PetriNetEditor
          initialContent={maimlContent}
          projectId={projectId}
          onSave={handleSave}
          onContentChange={() => setHasUnsavedChanges(true)}
        />
      </div>
    </div>
  );
}
```

### 3. ProjectManagement修正

```typescript
// frontend/src/components/project/ProjectManagement.tsx

interface ProjectManagementProps {
  onEdit?: (projectId: string) => void;  // 🆕 追加
}

export default function ProjectManagement({ onEdit }: ProjectManagementProps) {
  // 既存のステート
  const [selectedProject, setSelectedProject] = useState<any>(null);
  
  // 🆕 Draft関連のステート
  const [draftRestoreDialogOpen, setDraftRestoreDialogOpen] = useState(false);
  const [draftInfo, setDraftInfo] = useState<any>(null);

  // 🆕 編集ボタンクリック
  const handleEditButtonClick = useCallback(() => {
    if (!selectedProject) {
      message.warning('プロジェクトを選択してください');
      return;
    }

    // Draft チェック
    const draftKey = `petri-net-draft-${selectedProject.id}`;
    const draftData = localStorage.getItem(draftKey);
    
    if (draftData) {
      try {
        const draft = JSON.parse(draftData);
        const info = {
          savedAt: draft.savedAt || new Date().toISOString(),
          methodCount: draft.methods?.length || 0,
          nodeCount: draft.nodes?.length || 0
        };
        
        setDraftInfo(info);
        setDraftRestoreDialogOpen(true);
        return;
      } catch (error) {
        console.error('Draft parse error:', error);
      }
    }

    // Draftがない場合は直接エディタへ
    onEdit?.(selectedProject.id);
  }, [selectedProject, onEdit]);

  // 🆕 Draft復元ハンドラー
  const handleRestoreDraft = useCallback(() => {
    setDraftRestoreDialogOpen(false);
    // TODO: Draft復元してエディタ起動
    onEdit?.(selectedProject?.id);
  }, [selectedProject, onEdit]);

  // 🆕 新規作成/既存ファイルハンドラー
  const handleCreateNewOrExisting = useCallback(() => {
    setDraftRestoreDialogOpen(false);
    onEdit?.(selectedProject?.id);
  }, [selectedProject, onEdit]);

  return (
    <>
      {/* 既存のProjectManagementのJSX */}
      {/* ... */}
      
      {/* 編集ボタン（行737付近を修正） */}
      <Button
        icon={<EditOutlined />}
        onClick={handleEditButtonClick}  {/* ✅ ハンドラー変更 */}
        disabled={!selectedProject}
      >
        編集
      </Button>

      {/* 🆕 Draft復元確認ダイアログ */}
      {draftInfo && (
        <DraftRestoreDialog
          open={draftRestoreDialogOpen}
          onClose={() => setDraftRestoreDialogOpen(false)}
          onRestoreDraft={handleRestoreDraft}
          onCreateNew={handleCreateNewOrExisting}
          draftInfo={draftInfo}
        />
      )}
    </>
  );
}
```

```typescript
// frontend/src/components/viewer/DraftRestoreDialog.tsx

'use client';

import React from 'react';
import { Modal, Button, Space, Alert, Typography } from 'antd';
import { InfoCircleOutlined } from '@ant-design/icons';

const { Text } = Typography;

interface DraftRestoreDialogProps {
  open: boolean;
  onClose: () => void;
  onRestoreDraft: () => void;
  onCreateNew: () => void;
  draftInfo: {
    savedAt: string;
    methodCount: number;
    nodeCount: number;
  };
}

export default function DraftRestoreDialog({
  open,
  onClose,
  onRestoreDraft,
  onCreateNew,
  draftInfo
}: DraftRestoreDialogProps) {
  return (
    <Modal
      title={
        <Space>
          <InfoCircleOutlined style={{ color: '#1890ff' }} />
          <span>一時保存データの復元</span>
        </Space>
      }
      open={open}
      onCancel={onClose}
      footer={[
        <Button key="new" onClick={onCreateNew}>
          新規作成/既存ファイルを開く
        </Button>,
        <Button key="restore" type="primary" onClick={onRestoreDraft}>
          一時保存を復元する
        </Button>
      ]}
      width={500}
    >
      <Space direction="vertical" size="middle" style={{ width: '100%' }}>
        <Alert
          message="一時保存データが見つかりました"
          description={`最終保存: ${new Date(draftInfo.savedAt).toLocaleString('ja-JP')}`}
          type="info"
          showIcon
        />
        
        <div style={{ 
          padding: '16px', 
          background: '#f5f5f5', 
          borderRadius: '4px' 
        }}>
          <Space direction="vertical" size="small">
            <Text>
              <strong>Method数:</strong> {draftInfo.methodCount}
            </Text>
            <Text>
              <strong>ノード数:</strong> {draftInfo.nodeCount}
            </Text>
          </Space>
        </div>

        <Alert
          message="注意"
          description="一時保存を復元しない場合、最新の保存済みファイルまたは新規作成から開始します。"
          type="warning"
          showIcon
        />
      </Space>
    </Modal>
  );
}
```
