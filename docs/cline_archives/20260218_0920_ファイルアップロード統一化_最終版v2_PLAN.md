## 1. タスク概要

S3ファイルアップロード機能を統一化します。フロントエンドの`fileService.ts`を`fileAccessService.ts`に統合し、バックエンドは共通の1つのエンドポイントを提供します。

**重要**: 
- CSVアップロード機能は対象外です（処理内容が異なるため）
- **fileTypeパラメータは使用しません**
- 各画面で完全なファイルパス（例: `projects/project_001`）を生成してから送信

### 統合の方針

#### フロントエンド統合
- `fileService.ts` → `fileAccessService.ts` にリネーム/統合
- アップロードとダウンロードを1つのサービスで提供
- **各画面で完全なファイルパスを生成**（`projects/xxx`, `formats/xxx/yyy`等）

#### バックエンド統合
- **共通エンドポイント**: `POST /api/v1/files/upload`
- **fileTypeパラメータなし** - targetPathから判断
- 個別エンドポイント（`/projects/:id/files/upload`等）は作成しない
- 既存の`fileAccessService.ts`にアップロード機能を追加

### バックエンド実装

#### fileAccessService.ts への追加

```typescript
/**
 * ファイルをアップロード（単一または複数）
 * 
 * @param organizationId 組織ID
 * @param userId ユーザーID
 * @param uploadMode アップロードモード（single, bulk, folder）
 * @param targetPath ターゲットパス（各画面で完全なパスを生成、相対パス）
 * @param files アップロードするファイル配列
 * @param metadata 追加メタデータ
 * @returns アップロード結果
 */
async uploadFiles(
  organizationId: string,
  userId: string,
  uploadMode: 'single' | 'bulk' | 'folder',
  targetPath: string,
  files: Express.Multer.File[],
  metadata?: {
    projectId?: string;
    formatId?: string;
    batchName?: string;
    preserveFolderStructure?: boolean;
  }
): Promise<{
  uploadMode: string;
  files: Array<{
    fileId: string;
    fileName: string;
    filePath: string;
    s3Key: string;
    fileSize: number;
    contentType: string;
    uploadedAt: string;
  }>;
  summary: {
    totalFiles: number;
    successfulFiles: number;
    failedFiles: number;
    totalSize: number;
  };
}> {
  // 1. バリデーション
  if (!files || files.length === 0) {
    throw new ValidationError('ファイルが選択されていません');
  }

  if (uploadMode === 'single' && files.length > 1) {
    throw new ValidationError('単一ファイルモードでは1つのファイルのみアップロード可能です');
  }

  // 2. ターゲットパスの正規化
  const normalizedPath = targetPath.replace(/^\/+|\/+$/g, '');

  // 3. 各ファイルをアップロード
  const uploadedFiles = [];
  let totalSize = 0;

  for (const file of files) {
    try {
      // ファイルパスの構築
      const fileName = file.originalname;
      const filePath = `${normalizedPath}/${fileName}`;

      // storageServiceを使用してS3/Swiftにアップロード
      const s3Key = await this.storageService.uploadFile(
        organizationId,
        filePath,
        file.buffer,
        { contentType: file.mimetype }
      );

      // file_metadataテーブルに保存
      const fileId = await this.saveFileMetadata({
        organizationId,
        userId,
        fileName,
        filePath,
        s3Key,
        fileSize: file.size,
        contentType: file.mimetype,
        projectId: metadata?.projectId,
        formatId: metadata?.formatId,
        batchName: metadata?.batchName,
      });

      uploadedFiles.push({
        fileId,
        fileName,
        filePath,
        s3Key,
        fileSize: file.size,
        contentType: file.mimetype,
        uploadedAt: new Date().toISOString(),
      });

      totalSize += file.size;
    } catch (error) {
      console.error(`File upload failed: ${file.originalname}`, error);
      // エラーハンドリング: 個別ファイルの失敗は記録して継続
    }
  }

  return {
    uploadMode,
    files: uploadedFiles,
    summary: {
      totalFiles: files.length,
      successfulFiles: uploadedFiles.length,
      failedFiles: files.length - uploadedFiles.length,
      totalSize,
    },
  };
}
```

#### fileAccessController.ts への追加

```typescript
/**
 * ファイルアップロードエンドポイント
 * POST /api/v1/files/upload
 */
export async function uploadFiles(req: Request, res: Response): Promise<void> {
  const requestId = uuidv4();

  try {
    const { organizationId, userId } = req.user!;
    const { uploadMode, targetPath } = req.body;
    const files = req.files as Express.Multer.File[];

    // metadataをパース
    let metadata;
    if (req.body.metadata) {
      try {
        metadata = JSON.parse(req.body.metadata);
      } catch (error) {
        metadata = {};
      }
    }

    // Service層呼び出し
    const repository = new FileMetadataRepository();
    const accessValidator = new ProjectAccessValidator();
    const service = new FileAccessService(repository, accessValidator);

    const result = await service.uploadFiles(
      organizationId,
      userId,
      uploadMode,
      targetPath,
      files,
      metadata
    );

    res.status(200).json({
      success: true,
      data: result,
      meta: {
        timestamp: new Date().toISOString(),
        requestId,
      },
    });
  } catch (error: any) {
    // エラーハンドリング
    logError('File upload error', error, {
      action: 'file_upload',
      organizationId: req.user?.organizationId,
      targetPath: req.body?.targetPath,
      requestId,
    });

    let statusCode = 500;
    let errorCode = 'SYSTEM_ERROR';
    let errorMessage = 'システムエラーが発生しました';

    if (error.message) {
      if (error.message.includes('バリデーション') || error.message.includes('無効')) {
        statusCode = 400;
        errorCode = 'VALIDATION_ERROR';
        errorMessage = error.message;
      } else if (error.message.includes('権限')) {
        statusCode = 403;
        errorCode = 'PERMISSION_INSUFFICIENT';
        errorMessage = error.message;
      } else if (error.message.includes('見つかりません')) {
        statusCode = 404;
        errorCode = 'RESOURCE_NOT_FOUND';
        errorMessage = error.message;
      } else if (error.message.includes('ストレージ')) {
        statusCode = 500;
        errorCode = 'STORAGE_ERROR';
        errorMessage = error.message;
      }
    }

    res.status(statusCode).json({
      success: false,
      error: {
        code: errorCode,
        message: errorMessage,
      },
      meta: {
        timestamp: new Date().toISOString(),
        requestId,
      },
    });
  }
}
```

### フロントエンド実装

#### fileAccessService.ts（リネーム後）

```typescript
/**
 * ファイルアクセスサービス
 * アップロードとダウンロードの統一サービス
 */

export interface UploadFilesParams {
  uploadMode: UploadMode;
  targetPath: string; // 各画面で完全なパスを生成して渡す
  files: File[];
  metadata?: {
    projectId?: string;
    formatId?: string;
    batchName?: string;
    preserveFolderStructure?: boolean;
  };
  onProgress?: (progress: UploadProgress) => void;
}

export const fileAccessService = {
  async uploadFiles(params: UploadFilesParams): Promise<UploadResponse> {
    const {
      uploadMode,
      targetPath,
      files,
      metadata,
      onProgress,
    } = params;

    const formData = new FormData();
    formData.append('uploadMode', uploadMode);
    formData.append('targetPath', targetPath);

    files.forEach((file) => {
      formData.append('files', file);
    });

    if (metadata) {
      formData.append('metadata', JSON.stringify(metadata));
    }

    const response = await fileApiClient.post<ApiResponse<UploadResponseData>>(
      `/files/upload`, // 共通エンドポイント
      formData,
      {
        headers: {
          'Content-Type': 'multipart/form-data',
        },
        onUploadProgress: (progressEvent: AxiosProgressEvent) => {
          if (onProgress && progressEvent.total) {
            const percentage = Math.round(
              (progressEvent.loaded * 100) / progressEvent.total
            );
            
            onProgress({
              uploadedBytes: progressEvent.loaded,
              totalBytes: progressEvent.total,
              percentage,
              uploadedCount: 0,
              totalCount: files.length,
            });
          }
        },
      }
    );

    if (response.data.success) {
      return {
        success: true,
        data: response.data.data,
      };
    }

    throw new FileAccessServiceError('アップロードに失敗しました', 'UPLOAD_FAILED');
  },
};
```
