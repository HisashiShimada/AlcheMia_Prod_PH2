**環境情報:**
```yaml

アカウント: test
```

### 便利なヘルパースクリプト

```bash
# スクリプトファイルを作成
cat > ~/swift-helper.sh << 'EOF'
#!/bin/bash

if [ -z "$ST_AUTH" ]; then
    export ST_AUTH=http://localhost:8080/auth/v1.0
    export ST_USER=test:tester
    export ST_KEY=testing
fi

CONTAINER="mmdpf-files"

case $1 in
    ls)
        swift list $CONTAINER --prefix "$2"
        ;;
    up)
        swift upload $CONTAINER "$2" --object-name "$3"
        ;;
    down)
        swift download $CONTAINER "$2" -o "$3"
        ;;
    rm)
        swift delete $CONTAINER "$2"
        ;;
    stat)
        if [ -z "$2" ]; then
            swift stat $CONTAINER
        else
            swift stat $CONTAINER "$2"
        fi
        ;;
    *)
        echo "使用方法:"
        echo "  $0 ls [prefix]         # ファイル一覧"
        echo "  $0 up <local> <remote> # アップロード"
        echo "  $0 down <remote> <local> # ダウンロード"
        echo "  $0 rm <remote>         # 削除"
        echo "  $0 stat [object]       # 統計情報"
        ;;
esac
EOF

chmod +x ~/swift-helper.sh
echo "alias sft='~/swift-helper.sh'" >> ~/.bashrc
source ~/.bashrc
```

**使用例:**
```bash
# ファイル一覧
sft ls org_maiml_lab_001/projects/

# アップロード
sft up /tmp/file.txt org_maiml_lab_001/projects/file.txt

# ダウンロード
sft down org_maiml_lab_001/projects/file.txt /tmp/file.txt

# 削除
sft rm org_maiml_lab_001/projects/file.txt

# 統計情報
sft stat
```

## 方法2: TypeScript/Node.jsから操作

### StorageServiceの使用

mmdpfプロジェクトでは、`storageService`を使ってSwiftを操作します。

#### セットアップ

```typescript
import { getStorageService } from '@/services/storageService';

const storage = getStorageService();
```

#### 基本操作

```typescript
// 1. ファイルのアップロード
const organizationPrefix = 'maiml_lab_001';
const filePath = 'projects/sample_project/sample.maiml';
const fileContent = Buffer.from('<MAIML>...</MAIML>');

await storage.uploadFile(
  organizationPrefix,
  filePath,
  fileContent,
  { contentType: 'application/xml' }
);

// 2. ファイルのダウンロード
const downloadedContent = await storage.downloadFile(
  organizationPrefix,
  filePath
);

// 3. ファイルの存在確認
const exists = await storage.fileExists(
  organizationPrefix,
  filePath
);

// 4. ファイル一覧の取得
const files = await storage.listFiles(
  organizationPrefix,
  'projects/'
);

// 5. ファイルのメタデータ取得
const metadata = await storage.getFileMetadata(
  organizationPrefix,
  filePath
);

// 6. 署名付きURLの生成（1時間有効）
const signedUrl = await storage.getSignedUrl(
  organizationPrefix,
  filePath,
  3600
);

// 7. ファイルのコピー
await storage.copyFile(
  organizationPrefix,
  'projects/sample.maiml',
  'projects/archive/sample_v1.maiml'
);

// 8. ファイルの削除
await storage.deleteFile(
  organizationPrefix,
  filePath
);
```

#### 実際の使用例：プロジェクト作成

```typescript
// backend/src/services/projectService.ts の例

import { getStorageService } from './storageService';

async function createProject(
  organizationId: string,
  projectName: string,
  initialMaiml: string
) {
  const storage = getStorageService();
  
  // 組織プレフィックスを取得
  const orgPrefix = await getOrganizationPrefix(organizationId);
  
  // MAIMLファイルをアップロード
  const filePath = `projects/${projectName}/${projectName}.maiml`;
  await storage.uploadFile(
    orgPrefix,
    filePath,
    Buffer.from(initialMaiml),
    { contentType: 'application/xml' }
  );
  
  // READMEファイルをアップロード
  const readmePath = `projects/${projectName}/README.md`;
  const readmeContent = `# ${projectName}\n\nプロジェクト作成日: ${new Date().toISOString()}`;
  await storage.uploadFile(
    orgPrefix,
    readmePath,
    Buffer.from(readmeContent),
    { contentType: 'text/markdown' }
  );
  
  return {
    success: true,
    filePath: `org_${orgPrefix}/${filePath}`
  };
}
```

## 方法3: curlコマンド

### 基本操作

#### 1. 認証トークンの取得

```bash
# 認証リクエスト
AUTH_RESPONSE=$(curl -i -s \
  -H "X-Auth-User: test:tester" \
  -H "X-Auth-Key: testing" \
  http://localhost:8080/auth/v1.0)

# トークンとストレージURLを抽出
AUTH_TOKEN=$(echo "$AUTH_RESPONSE" | grep -i "X-Auth-Token:" | awk '{print $2}' | tr -d '\r')
STORAGE_URL=$(echo "$AUTH_RESPONSE" | grep -i "X-Storage-Url:" | awk '{print $2}' | tr -d '\r')

echo "Auth Token: $AUTH_TOKEN"
echo "Storage URL: $STORAGE_URL"
```

#### 2. コンテナ一覧を取得

```bash
curl -H "X-Auth-Token: $AUTH_TOKEN" $STORAGE_URL
```

#### 3. ファイルをアップロード

```bash
echo "Hello from curl!" > /tmp/test-upload.txt

curl -X PUT \
  -H "X-Auth-Token: $AUTH_TOKEN" \
  -T /tmp/test-upload.txt \
  $STORAGE_URL/mmdpf-files/org_maiml_lab_001/projects/test.txt
```

#### 4. ファイル一覧を取得

```bash
# 全ファイル
curl -H "X-Auth-Token: $AUTH_TOKEN" \
  $STORAGE_URL/mmdpf-files

# プレフィックスで絞り込み
curl -H "X-Auth-Token: $AUTH_TOKEN" \
  "$STORAGE_URL/mmdpf-files?prefix=org_maiml_lab_001/projects/"
```

#### 5. ファイルをダウンロード

```bash
curl -H "X-Auth-Token: $AUTH_TOKEN" \
  $STORAGE_URL/mmdpf-files/org_maiml_lab_001/projects/test.txt \
  -o /tmp/test-download.txt
```

#### 6. ファイルを削除

```bash
curl -X DELETE \
  -H "X-Auth-Token: $AUTH_TOKEN" \
  $STORAGE_URL/mmdpf-files/org_maiml_lab_001/projects/test.txt
```

### ヘルパースクリプト

```bash
cat > ~/swift-curl.sh << 'EOF'
#!/bin/bash

AUTH_URL="http://localhost:8080/auth/v1.0"
AUTH_USER="test:tester"
AUTH_KEY="testing"
CONTAINER="mmdpf-files"

get_auth() {
    RESPONSE=$(curl -i -s -H "X-Auth-User: $AUTH_USER" -H "X-Auth-Key: $AUTH_KEY" $AUTH_URL)
    AUTH_TOKEN=$(echo "$RESPONSE" | grep -i "X-Auth-Token:" | awk '{print $2}' | tr -d '\r')
    STORAGE_URL=$(echo "$RESPONSE" | grep -i "X-Storage-Url:" | awk '{print $2}' | tr -d '\r')
}

case $1 in
    ls)
        get_auth
        if [ -z "$2" ]; then
            curl -s -H "X-Auth-Token: $AUTH_TOKEN" $STORAGE_URL/$CONTAINER
        else
            curl -s -H "X-Auth-Token: $AUTH_TOKEN" "$STORAGE_URL/$CONTAINER?prefix=$2"
        fi
        ;;
    up)
        get_auth
        curl -X PUT -H "X-Auth-Token: $AUTH_TOKEN" -T "$2" \
            $STORAGE_URL/$CONTAINER/$3
        echo "✅ アップロード完了: $3"
        ;;
    down)
        get_auth
        curl -H "X-Auth-Token: $AUTH_TOKEN" \
            $STORAGE_URL/$CONTAINER/$2 -o "$3"
        echo "✅ ダウンロード完了: $3"
        ;;
    rm)
        get_auth
        curl -X DELETE -H "X-Auth-Token: $AUTH_TOKEN" \
            $STORAGE_URL/$CONTAINER/$2
        echo "✅ 削除完了: $2"
        ;;
    *)
        echo "使用方法: $0 {ls|up|down|rm} [args]"
        ;;
esac
EOF

chmod +x ~/swift-curl.sh
echo "alias sftc='~/swift-curl.sh'" >> ~/.bashrc
source ~/.bashrc
```

## トラブルシューティング

### 問題1: 接続できない

**症状:**
```
Connection refused
```

**対策:**

```bash
# Swiftコンテナが起動しているか確認
docker-compose ps swift

# 起動していない場合
docker-compose up -d swift

# ログを確認
docker-compose logs swift --tail 50
```

### 問題2: 認証エラー

**症状:**
```
401 Unauthorized
```

**対策:**

```bash
# 環境変数を確認
echo $ST_AUTH
echo $ST_USER
echo $ST_KEY

# 再設定
export ST_AUTH=http://localhost:8080/auth/v1.0
export ST_USER=test:tester
export ST_KEY=testing

# 手動で認証テスト
curl -i -H "X-Auth-User: test:tester" \
  -H "X-Auth-Key: testing" \
  http://localhost:8080/auth/v1.0
```

### 問題3: ファイルが見つからない

**症状:**
```
404 Not Found
```

**対策:**

```bash
# ファイルパスを確認
sft ls org_maiml_lab_001/projects/

# organization_prefix が正しいか確認
# 正: org_maiml_lab_001
# 誤: maiml_lab_001 (org_ プレフィックスなし)
```

### 問題4: コンテナが存在しない

**症状:**
```
Container not found
```

**対策:**

```bash
# コンテナを作成
docker exec -it mmdpf-swift \
  swift -A http://localhost:8080/auth/v1.0 \
  -U test:tester -K testing post mmdpf-files

# 確認
swift list
```

### 問題5: パーミッションエラー

**症状:**
```
Permission denied
```

**対策:**

マルチテナント分離により、他の組織のファイルにはアクセスできません。

```bash
# ✅ 正しい: 自組織のファイル
sft ls org_maiml_lab_001/projects/

# ❌ 間違い: 他組織のファイル（アクセス不可）
# ※ TypeScriptコード内でアクセス制御されている
```

## ベストプラクティス

### DO（推奨）

✅ 常に `organization_prefix` を含める
✅ ファイルパスは相対パスで指定
✅ アーカイブフォルダを使って履歴管理
✅ 署名付きURLを使ってセキュアにアクセス
✅ エラーハンドリングを実装
✅ ログを記録する

### DON'T（非推奨）

❌ `org_` プレフィックスなしでアクセス
❌ ハードコーディングされたパス
❌ organization_idチェックを省略
❌ エラーを無視する
❌ 大量のファイルを同期的に処理
