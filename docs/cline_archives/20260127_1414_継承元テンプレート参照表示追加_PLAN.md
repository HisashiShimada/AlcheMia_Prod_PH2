## 5. 技術的詳細

### 継承元テンプレート取得
```typescript
const inheritedTemplate = useMemo(() => {
  // templateRefIdから継承元を検索
  const templateRefId = node?.data?.templateRefId;
  if (!templateRefId) return null;
  
  // allNodesから該当するテンプレートを検索
  return allNodes.find(n => {
    const template = n.data?.templateRef || n.data?.template;
    return template?.id === templateRefId;
  });
}, [node, allNodes]);
```

### 継承元情報の抽出
```typescript
const inheritedInfo = useMemo(() => {
  if (!inheritedTemplate) return null;
  
  const template = inheritedTemplate.data?.templateRef || inheritedTemplate.data?.template;
  
  return {
    name: template?.name || inheritedTemplate.data?.name,
    description: template?.description || inheritedTemplate.data?.description,
    uuid: template?.uuid,
    properties: template?.properties || {},
    contents: template?.contents || {},
    nodeId: inheritedTemplate.id
  };
}, [inheritedTemplate]);
```

### プロパティの簡易表示
```typescript
// 再帰的なネスト構造ではなく、簡易表示
const renderPropertySummary = (propData: PropertyData): string => {
  const attrs = Object.entries(propData.attributes || {})
    .map(([k, v]) => `${k}="${v}"`)
    .join(', ');
  
  const elems = Object.entries(propData.elements || {})
    .map(([k, v]) => `${k}: ${v}`)
    .join(', ');
  
  return [attrs, elems].filter(Boolean).join(' | ');
};
```
