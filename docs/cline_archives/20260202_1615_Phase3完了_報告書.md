1. **formatTransitionDisplayName()ヘルパーメソッド追加**
   ```typescript
   private formatTransitionDisplayName(
     id: string,
     name: string | undefined,
     methodId: string
   ): string {
     const methodPrefix = methodId && methodId !== 'method_0' 
       ? `[${methodId}] ` 
       : '';
     
     // 名前が長すぎる場合は省略（50文字超）
     if (name && name.length > 50) {
       return `${methodPrefix}${id} - ${name.substring(0, 47)}...`;
     }
     
     // 名前がある場合は "ID - 名前" 形式
     if (name) {
       return `${methodPrefix}${id} - ${name}`;
     }
     
     // 名前がない場合はIDのみ
     return `${methodPrefix}${id}`;
   }
   ```
