## 1. 設計概要

### 1.1 設計方針
- **最小限の変更**: 既存のexpress-validatorバリデーション定義を最大限活用
- **段階的移行**: 既存コードへの影響を最小化しながら段階的に適用
- **統一されたエラーレスポンス**: api.mdで定義された形式に準拠
- **再利用性**: 共通バリデーター関数で重複を削減

## 3. 詳細設計

### 3.1 handleValidationErrors ミドルウェア

#### 3.1.1 ファイル: `middleware/validationMiddleware.ts`

```typescript
/**
 * バリデーションエラーハンドリングミドルウェア
 * 
 * @description express-validatorのバリデーション結果をチェックし、
 *              エラーがあれば統一されたレスポンス形式で返却する
 */

import { Request, Response, NextFunction } from 'express';
import { validationResult, ValidationError } from 'express-validator';
import { v4 as uuidv4 } from 'uuid';

/**
 * バリデーションエラー詳細の型定義
 */
interface ValidationErrorDetail {
  field: string;
  message: string;
  code: string;
  value?: any;
}

/**
 * バリデーションエラーハンドリングミドルウェア
 * 
 * @param req - Expressリクエスト
 * @param res - Expressレスポンス
 * @param next - 次のミドルウェア
 */
export const handleValidationErrors = (
  req: Request,
  res: Response,
  next: NextFunction
): void => {
  const errors = validationResult(req);

  if (!errors.isEmpty()) {
    // バリデーションエラーを統一形式に変換
    const details: ValidationErrorDetail[] = errors.array().map((error: ValidationError) => {
      // エラーの種類に応じてフィールド名を取得
      let field = 'unknown';
      let value: any;

      if (error.type === 'field') {
        field = error.path;
        value = error.value;
      }

      return {
        field,
        message: error.msg,
        code: mapErrorToCode(error),
        ...(value !== undefined && { value }),
      };
    });

    // 統一されたエラーレスポンスを返却
    const requestId = (req.headers['x-request-id'] as string) || uuidv4();

    res.status(400).json({
      success: false,
      error: {
        code: 'VALIDATION_ERROR',
        message: 'バリデーションエラーが発生しました',
        details,
      },
      meta: {
        timestamp: new Date().toISOString(),
        requestId,
      },
    });
    return;
  }

  // バリデーションエラーがない場合は次のミドルウェアへ
  next();
};

/**
 * express-validatorのエラーを適切なエラーコードにマッピング
 * 
 * @param error - ValidationError
 * @returns エラーコード文字列
 */
function mapErrorToCode(error: ValidationError): string {
  if (error.type !== 'field') {
    return 'VALIDATION_ERROR';
  }

  // msgから推測してコードを決定
  const msg = error.msg.toLowerCase();

  if (msg.includes('required') || msg.includes('必須')) {
    return 'REQUIRED_FIELD';
  }
  if (msg.includes('length') || msg.includes('文字')) {
    return 'STRING_LENGTH_INVALID';
  }
  if (msg.includes('uuid') || msg.includes('format') || msg.includes('形式')) {
    return 'INVALID_FORMAT';
  }
  if (msg.includes('integer') || msg.includes('整数')) {
    return 'INVALID_TYPE';
  }
  if (msg.includes('range') || msg.includes('範囲')) {
    return 'OUT_OF_RANGE';
  }
  if (msg.includes('invalid') || msg.includes('不正')) {
    return 'INVALID_VALUE';
  }

  return 'VALIDATION_ERROR';
}

/**
 * カスタムエラーコードを指定したバリデーションエラーレスポンスを生成
 * 
 * @param field - フィールド名
 * @param message - エラーメッセージ
 * @param code - エラーコード
 * @param value - エラーが発生した値（オプション）
 */
export const createValidationError = (
  field: string,
  message: string,
  code: string = 'VALIDATION_ERROR',
  value?: any
): ValidationErrorDetail => {
  return {
    field,
    message,
    code,
    ...(value !== undefined && { value }),
  };
};
```

#### 3.1.2 責務
1. **validationResult()の実行**: express-validatorの結果を取得
2. **エラー形式の統一**: ValidationErrorを統一形式に変換
3. **エラーコードのマッピング**: メッセージからエラーコードを推測
4. **レスポンスの返却**: 400 Bad Requestで統一形式のJSONを返す

#### 3.1.3 特徴
- エラーがない場合は`next()`を呼び出して次のミドルウェアへ
- requestIdはリクエストヘッダーから取得、なければ生成
- エラーコードは自動的にマッピング（カスタマイズ可能）

### 3.2 共通バリデーター関数

#### 3.2.1 ファイル: `utils/validators.ts`

```typescript
/**
 * 共通バリデーター関数集
 * 
 * @description 再利用可能なバリデーションルールを定義
 */

import { param, body, query, ValidationChain } from 'express-validator';

/**
 * UUID検証
 * 
 * @param field - フィールド名
 * @param location - パラメータの場所 ('param' | 'body' | 'query')
 * @param options - オプション設定
 * @returns ValidationChain
 */
export const validateUUID = (
  field: string,
  location: 'param' | 'body' | 'query' = 'param',
  options: { optional?: boolean } = {}
): ValidationChain => {
  const validator = location === 'param' ? param(field) :
                   location === 'body' ? body(field) :
                   query(field);

  let chain = validator;

  if (options.optional) {
    chain = chain.optional();
  }

  return chain
    .isUUID()
    .withMessage(`${field}はUUID形式である必要があります`);
};

/**
 * 必須文字列検証（長さ制限付き）
 * 
 * @param field - フィールド名
 * @param minLength - 最小文字数
 * @param maxLength - 最大文字数
 * @param location - パラメータの場所
 * @returns ValidationChain
 */
export const validateRequiredString = (
  field: string,
  minLength: number = 1,
  maxLength: number = 255,
  location: 'body' | 'query' = 'body'
): ValidationChain => {
  const validator = location === 'body' ? body(field) : query(field);

  return validator
    .notEmpty()
    .withMessage(`${field}は必須です`)
    .isString()
    .withMessage(`${field}は文字列である必要があります`)
    .isLength({ min: minLength, max: maxLength })
    .withMessage(`${field}は${minLength}〜${maxLength}文字である必要があります`);
};

/**
 * オプショナル文字列検証（長さ制限付き）
 * 
 * @param field - フィールド名
 * @param maxLength - 最大文字数
 * @param location - パラメータの場所
 * @returns ValidationChain
 */
export const validateOptionalString = (
  field: string,
  maxLength: number = 1000,
  location: 'body' | 'query' = 'body'
): ValidationChain => {
  const validator = location === 'body' ? body(field) : query(field);

  return validator
    .optional()
    .isString()
    .withMessage(`${field}は文字列である必要があります`)
    .isLength({ max: maxLength })
    .withMessage(`${field}は${maxLength}文字以内である必要があります`);
};

/**
 * 列挙型検証
 * 
 * @param field - フィールド名
 * @param allowedValues - 許可された値の配列
 * @param location - パラメータの場所
 * @param options - オプション設定
 * @returns ValidationChain
 */
export const validateEnum = (
  field: string,
  allowedValues: string[],
  location: 'query' | 'body' = 'query',
  options: { optional?: boolean } = { optional: true }
): ValidationChain => {
  const validator = location === 'query' ? query(field) : body(field);

  let chain = validator;

  if (options.optional) {
    chain = chain.optional();
  }

  return chain
    .isIn(allowedValues)
    .withMessage(`${field}は次のいずれかである必要があります: ${allowedValues.join(', ')}`);
};

/**
 * 整数範囲検証
 * 
 * @param field - フィールド名
 * @param min - 最小値
 * @param max - 最大値（オプション）
 * @param location - パラメータの場所
 * @param options - オプション設定
 * @returns ValidationChain
 */
export const validateIntRange = (
  field: string,
  min: number,
  max?: number,
  location: 'query' | 'body' = 'query',
  options: { optional?: boolean } = { optional: true }
): ValidationChain => {
  const validator = location === 'query' ? query(field) : body(field);

  let chain = validator;

  if (options.optional) {
    chain = chain.optional();
  }

  chain = chain.isInt({ min, ...(max && { max }) });

  if (max !== undefined) {
    chain = chain.withMessage(`${field}は${min}〜${max}の範囲の整数である必要があります`);
  } else {
    chain = chain.withMessage(`${field}は${min}以上の整数である必要があります`);
  }

  return chain;
};

/**
 * Boolean型検証
 * 
 * @param field - フィールド名
 * @param location - パラメータの場所
 * @returns ValidationChain
 */
export const validateBoolean = (
  field: string,
  location: 'query' | 'body' = 'query'
): ValidationChain => {
  const validator = location === 'query' ? query(field) : body(field);

  return validator
    .optional()
    .isBoolean()
    .withMessage(`${field}はtrueまたはfalseである必要があります`);
};

/**
 * 制御文字チェック付き文字列検証
 * 
 * @param field - フィールド名
 * @param minLength - 最小文字数
 * @param maxLength - 最大文字数
 * @returns ValidationChain
 */
export const validateStringNoControlChars = (
  field: string,
  minLength: number = 1,
  maxLength: number = 255
): ValidationChain => {
  return body(field)
    .notEmpty()
    .withMessage(`${field}は必須です`)
    .isLength({ min: minLength, max: maxLength })
    .withMessage(`${field}は${minLength}〜${maxLength}文字である必要があります`)
    .matches(/^[^\x00-\x1F\x7F]*$/)
    .withMessage(`${field}に制御文字を含めることはできません`);
};

/**
 * 英数字+記号のみ検証
 * 
 * @param field - フィールド名
 * @param minLength - 最小文字数
 * @param maxLength - 最大文字数
 * @returns ValidationChain
 */
export const validateAlphanumericWithSymbols = (
  field: string,
  minLength: number = 1,
  maxLength: number = 100
): ValidationChain => {
  return body(field)
    .notEmpty()
    .withMessage(`${field}は必須です`)
    .isLength({ min: minLength, max: maxLength })
    .withMessage(`${field}は${minLength}〜${maxLength}文字である必要があります`)
    .matches(/^[a-zA-Z0-9_-]+$/)
    .withMessage(`${field}は英数字、ハイフン、アンダースコアのみ使用できます`);
};

/**
 * PINコード検証（6桁の数字）
 * 
 * @param field - フィールド名
 * @returns ValidationChain
 */
export const validatePinCode = (field: string = 'pinCode'): ValidationChain => {
  return body(field)
    .notEmpty()
    .withMessage('PINコードは必須です')
    .matches(/^\d{6}$/)
    .withMessage('PINコードは6桁の数字である必要があります');
};

/**
 * パスワード検証
 * 
 * @param field - フィールド名
 * @param minLength - 最小文字数
 * @param maxLength - 最大文字数
 * @returns ValidationChain
 */
export const validatePassword = (
  field: string = 'password',
  minLength: number = 8,
  maxLength: number = 255
): ValidationChain => {
  return body(field)
    .notEmpty()
    .withMessage('パスワードは必須です')
    .isLength({ min: minLength, max: maxLength })
    .withMessage(`パスワードは${minLength}〜${maxLength}文字である必要があります`);
};
```

```typescript
/**
 * バリデーション関連の型定義
 */

import { ValidationChain } from 'express-validator';

/**
 * バリデーションエラー詳細
 */
export interface ValidationErrorDetail {
  field: string;
  message: string;
  code: string;
  value?: any;
}

/**
 * バリデーションエラーレスポンス
 */
export interface ValidationErrorResponse {
  success: false;
  error: {
    code: 'VALIDATION_ERROR';
    message: string;
    details: ValidationErrorDetail[];
  };
  meta: {
    timestamp: string;
    requestId: string;
  };
}

/**
 * バリデーションルールセット
 */
export type ValidationRules = ValidationChain[];

/**
 * バリデーションオプション
 */
export interface ValidationOptions {
  optional?: boolean;
  custom?: (value: any) => boolean | Promise<boolean>;
  customErrorMessage?: string;
}

/**
 * フィールドバリデーション設定
 */
export interface FieldValidation {
  field: string;
  rules: ValidationRules;
  location?: 'body' | 'query' | 'param';
}
```
