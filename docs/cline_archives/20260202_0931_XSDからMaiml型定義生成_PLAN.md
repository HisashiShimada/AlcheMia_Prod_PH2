### XSD → TypeScript 変換マッピング

```typescript
// XSD complexType → TypeScript interface
<xs:complexType name="rootObjectType">
  <xs:sequence>
    <xs:element name="document" type="documentType" />
  </xs:sequence>
  <xs:attribute name="version" use="required" type="xs:string" />
</xs:complexType>

↓

export interface RootObjectType {
  document: DocumentType;
  version: string; // required
  features?: string; // optional
}
```

```typescript
// XSD simpleType → TypeScript type alias
<xs:simpleType name="uuid">
  <xs:restriction base="xs:string">
    <xs:pattern value="[a-f0-9]{8}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{12}"/>
  </xs:restriction>
</xs:simpleType>

↓

export type Uuid = string; // Pattern: [a-f0-9]{8}-...
```

```typescript
// XSD choice → TypeScript Union
<xs:choice>
  <xs:element name="material" type="materialType"/>
  <xs:element name="condition" type="conditionType"/>
</xs:choice>

↓

export type PlaceContent = MaterialType | ConditionType;
```

## 11. 実装例

### 生成ツールの基本構造

```typescript
// backend/scripts/generate-types-from-xsd.ts
import * as fs from 'fs/promises';
import * as path from 'path';
import { parseStringPromise } from 'xml2js';

interface GeneratorConfig {
  schemasDir: string;
  outputFile: string;
  namespace: string;
}

class MaimlTypeGenerator {
  private config: GeneratorConfig;
  private schemas: Map<string, any> = new Map();
  private generatedTypes: string[] = [];

  constructor(config: GeneratorConfig) {
    this.config = config;
  }

  async generate(): Promise<void> {
    console.log('🔍 XSDファイルを読み込み中...');
    await this.loadSchemas();
    
    console.log('🔨 型定義を生成中...');
    await this.generateTypes();
    
    console.log('💾 ファイルに書き込み中...');
    await this.writeOutput();
    
    console.log('✅ 型定義生成完了！');
  }

  private async loadSchemas(): Promise<void> {
    const schemaFiles = [
      'maiml.xsd',
      'maiml-core.xsd',
      'maiml-simpleTypes.xsd',
      'maiml-property.xsd',
      // ... 他のXSDファイル
    ];

    for (const file of schemaFiles) {
      const filePath = path.join(this.config.schemasDir, file);
      const content = await fs.readFile(filePath, 'utf-8');
      const parsed = await parseStringPromise(content);
      this.schemas.set(file, parsed);
    }
  }

  private async generateTypes(): Promise<void> {
    for (const [filename, schema] of this.schemas) {
      console.log(`  - ${filename} を処理中...`);
      this.processSchema(schema);
    }
  }

  private processSchema(schema: any): void {
    const complexTypes = schema['xs:schema']?.['xs:complexType'] || [];
    const simpleTypes = schema['xs:schema']?.['xs:simpleType'] || [];

    for (const complexType of complexTypes) {
      const typeDef = this.generateInterface(complexType);
      if (typeDef) {
        this.generatedTypes.push(typeDef);
      }
    }

    for (const simpleType of simpleTypes) {
      const typeDef = this.generateTypeAlias(simpleType);
      if (typeDef) {
        this.generatedTypes.push(typeDef);
      }
    }
  }

  private generateInterface(complexType: any): string | null {
    const name = complexType.$?.name;
    if (!name) return null;

    const interfaceName = this.toPascalCase(name);
    let code = `/**\n * ${name}\n * @generated from XSD\n */\n`;
    code += `export interface ${interfaceName} {\n`;

    // sequence要素の処理
    const sequence = complexType['xs:sequence']?.[0];
    if (sequence?.['xs:element']) {
      for (const element of sequence['xs:element']) {
        const propName = element.$?.name;
        const propType = this.mapXsdTypeToTs(element.$?.type || 'any');
        const optional = element.$?.minOccurs === '0';
        
        code += `  ${propName}${optional ? '?' : ''}: ${propType};\n`;
      }
    }

    // attribute要素の処理
    const attributes = complexType['xs:attribute'] || [];
    for (const attr of attributes) {
      const attrName = attr.$?.name;
      const attrType = this.mapXsdTypeToTs(attr.$?.type || 'string');
      const required = attr.$?.use === 'required';
      
      code += `  ${attrName}${required ? '' : '?'}: ${attrType};\n`;
    }

    code += '}\n\n';
    return code;
  }

  private generateTypeAlias(simpleType: any): string | null {
    const name = simpleType.$?.name;
    if (!name) return null;

    const aliasName = this.toPascalCase(name);
    const restriction = simpleType['xs:restriction']?.[0];
    const baseType = restriction?.$?.base || 'string';
    const tsType = this.mapXsdTypeToTs(baseType);

    let code = `/**\n * ${name}\n * @generated from XSD\n */\n`;
    code += `export type ${aliasName} = ${tsType};\n\n`;
    
    return code;
  }

  private mapXsdTypeToTs(xsdType: string): string {
    const typeMap: Record<string, string> = {
      'xs:string': 'string',
      'xs:int': 'number',
      'xs:integer': 'number',
      'xs:decimal': 'number',
      'xs:float': 'number',
      'xs:double': 'number',
      'xs:boolean': 'boolean',
      'xs:date': 'string',
      'xs:dateTime': 'string',
      'xs:ID': 'string',
      'xs:IDREF': 'string',
      'xs:QName': 'string',
      'xs:anyURI': 'string',
      'xs:base64Binary': 'string',
    };

    return typeMap[xsdType] || this.toPascalCase(xsdType.replace(/Type$/, ''));
  }

  private toPascalCase(str: string): string {
    return str
      .split(/[-_]/)
      .map(word => word.charAt(0).toUpperCase() + word.slice(1))
      .join('');
  }

  private async writeOutput(): Promise<void> {
    const header = `/**
 * MAIML Schema Type Definitions
 * 
 * @generated from XSD schemas in backend/reference/schemas/
 * @do-not-edit This file is automatically generated
 * @version 1.0.0
 * @date ${new Date().toISOString()}
 */

/* eslint-disable @typescript-eslint/no-explicit-any */

`;

    const content = header + this.generatedTypes.join('');
    await fs.writeFile(this.config.outputFile, content, 'utf-8');
  }
}

// 実行
const generator = new MaimlTypeGenerator({
  schemasDir: path.join(__dirname, '../../reference/schemas'),
  outputFile: path.join(__dirname, '../../../frontend/src/types/maiml-schema.ts'),
  namespace: 'MaimlSchema',
});

generator.generate().catch(console.error);
```
