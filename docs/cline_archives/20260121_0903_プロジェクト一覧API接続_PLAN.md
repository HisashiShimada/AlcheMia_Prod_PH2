## 5. 技術的詳細

### バックエンドAPI
- **エンドポイント**: `GET /api/v1/projects`
- **認証**: Cookie認証（HTTPOnly JWT）
- **レスポンス形式**:
```typescript
{
  success: true,
  data: {
    projects: [
      {
        id: string,
        project_name: string,
        description: string,
        status: string,
        created_at: string,
        updated_at: string,
        is_checked_out: boolean,
        checked_out_by: string | null,
        checked_out_at: string | null,
        organization_id: string,
        organization_name: string,
        owner: string,
        department: string,
        tags: string[],
      }
    ]
  },
  meta: {
    timestamp: string,
    requestId: string
  }
}
```

### フロントエンド実装
```typescript
export async function fetchProjects(): Promise<ApiResponse<ProjectListResponse>> {
  try {
    const response = await fetch(`${API_BASE_URL}/projects`, {
      method: 'GET',
      credentials: 'include', // Cookie認証
      headers: {
        'Content-Type': 'application/json',
      },
    });

    if (!response.ok) {
      const errorData = await response.json().catch(() => ({}));
      throw new Error(
        errorData.error?.message || 'プロジェクト一覧の取得に失敗しました'
      );
    }

    const data: ApiResponse<ProjectListResponse> = await response.json();
    return data;
  } catch (error) {
    console.error('fetchProjects error:', error);
    throw error;
  }
}
```
