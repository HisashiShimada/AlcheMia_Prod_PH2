```typescript
// POST /api/v1/auth/login
Request:
{
  username: string;
  password: string;
}
Headers:
{
  'X-Organization-Id': string;
}

Response (Success):
{
  success: true;
  data: {
    sessionToken: string;
    message: string;
    email: string;
    expiresIn: number;
  };
  meta: {
    timestamp: string;
    requestId: string;
  };
}

Response (Error):
{
  success: false;
  error: {
    code: string;
    message: string;
    details?: Array<{
      field: string;
      message: string;
      code: string;
    }>;
  };
  meta: {
    timestamp: string;
    requestId: string;
  };
}
```
