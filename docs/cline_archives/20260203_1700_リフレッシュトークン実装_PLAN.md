### Phase 2: JWTサービスの拡張

```typescript
/**
 * アクセストークンを生成（15分）
 */
async generateAccessToken(
  payload: Omit<JwtPayload, 'iat' | 'exp'>
): Promise<Result<string, ServiceError>> {
  const now = Math.floor(Date.now() / 1000);
  const expiresIn = parseInt(process.env.JWT_ACCESS_TOKEN_EXPIRES_IN || '900', 10); // 15分
  
  const fullPayload: JwtPayload = {
    ...payload,
    iat: now,
    exp: now + expiresIn
  };

  // HS256署名でJWT生成
  const jwtToken = jwt.sign(fullPayload, this.secret, {
    algorithm: 'HS256',
    issuer: this.issuer,
    audience: this.audience
  });

  // トークン形式: maiml_session-{sessionId}_{timestamp}.{jwt}
  const sessionToken = `maiml_session-${payload.sessionId}_${now}.${jwtToken}`;

  return { success: true, data: sessionToken };
}

/**
 * リフレッシュトークンを生成（7日間）
 */
async generateRefreshToken(
  userId: string,
  organizationId: string,
  sessionId: string
): Promise<Result<string, ServiceError>> {
  const now = Math.floor(Date.now() / 1000);
  const expiresIn = parseInt(process.env.JWT_REFRESH_TOKEN_EXPIRES_IN || '604800', 10); // 7日
  const tokenId = crypto.randomUUID();

  const payload: RefreshTokenPayload = {
    userId,
    organizationId,
    sessionId,
    tokenId,
    iat: now,
    exp: now + expiresIn
  };

  const jwtToken = jwt.sign(payload, this.secret, {
    algorithm: 'HS256',
    issuer: this.issuer,
    audience: this.audience
  });

  return { success: true, data: jwtToken };
}

/**
 * トークンペアを生成
 */
async generateTokenPair(
  payload: Omit<JwtPayload, 'iat' | 'exp'>
): Promise<Result<TokenPair, ServiceError>> {
  const accessTokenResult = await this.generateAccessToken(payload);
  if (!accessTokenResult.success) {
    return accessTokenResult;
  }

  const refreshTokenResult = await this.generateRefreshToken(
    payload.userId,
    payload.organizationId,
    payload.sessionId
  );
  if (!refreshTokenResult.success) {
    return refreshTokenResult;
  }

  return {
    success: true,
    data: {
      accessToken: accessTokenResult.data,
      refreshToken: refreshTokenResult.data
    }
  };
}

/**
 * リフレッシュトークンを検証
 */
async verifyRefreshToken(
  token: string
): Promise<Result<RefreshTokenPayload, ServiceError>> {
  try {
    const decoded = jwt.verify(token, this.secret, {
      algorithms: ['HS256'],
      issuer: this.issuer,
      audience: this.audience
    }) as RefreshTokenPayload;

    return { success: true, data: decoded };
  } catch (error) {
    if (error instanceof jwt.TokenExpiredError) {
      return {
        success: false,
        error: {
          code: 'REFRESH_TOKEN_EXPIRED',
          message: 'Refresh token has expired',
          originalError: error
        }
      };
    }

    return {
      success: false,
      error: {
        code: 'REFRESH_TOKEN_INVALID',
        message: 'Invalid refresh token',
        originalError: error as Error
      }
    };
  }
}
```

### Phase 3: Redisトークン管理サービスの作成

#### ファイル: `backend/src/services/refreshTokenService.ts`（新規）

**機能:**
1. リフレッシュトークンをRedisに保存
2. リフレッシュトークンをRedisから取得
3. リフレッシュトークンを無効化（削除）
4. ユーザーの全トークンを無効化
5. 期限切れトークンの自動削除

```typescript
import { getRedisClient } from '../config/redis';
import type { StoredRefreshToken, RefreshTokenPayload } from '../types/auth';

export class RefreshTokenService {
  private readonly prefix = 'mmdpf:refresh_token:';

  /**
   * リフレッシュトークンをRedisに保存
   */
  async save(
    payload: RefreshTokenPayload,
    ipAddress?: string,
    userAgent?: string
  ): Promise<void> {
    const redisClient = getRedisClient();
    if (!redisClient) {
      throw new Error('Redis client is not available');
    }

    const key = `${this.prefix}${payload.tokenId}`;
    const ttl = payload.exp - Math.floor(Date.now() / 1000);

    const data: StoredRefreshToken = {
      userId: payload.userId,
      organizationId: payload.organizationId,
      sessionId: payload.sessionId,
      tokenId: payload.tokenId,
      issuedAt: payload.iat,
      expiresAt: payload.exp,
      ipAddress,
      userAgent
    };

    // TTL付きで保存（期限切れ時に自動削除）
    await redisClient.setEx(key, ttl, JSON.stringify(data));

    // ユーザーごとのトークンリストに追加
    const userKey = `${this.prefix}user:${payload.userId}`;
    await redisClient.sAdd(userKey, payload.tokenId);
    await redisClient.expire(userKey, ttl);
  }

  /**
   * リフレッシュトークンをRedisから取得
   */
  async get(tokenId: string): Promise<StoredRefreshToken | null> {
    const redisClient = getRedisClient();
    if (!redisClient) {
      return null;
    }

    const key = `${this.prefix}${tokenId}`;
    const data = await redisClient.get(key);

    if (!data) {
      return null;
    }

    return JSON.parse(data) as StoredRefreshToken;
  }

  /**
   * リフレッシュトークンを無効化（削除）
   */
  async revoke(tokenId: string, userId: string): Promise<void> {
    const redisClient = getRedisClient();
    if (!redisClient) {
      return;
    }

    const key = `${this.prefix}${tokenId}`;
    await redisClient.del(key);

    // ユーザーのトークンリストからも削除
    const userKey = `${this.prefix}user:${userId}`;
    await redisClient.sRem(userKey, tokenId);
  }

  /**
   * ユーザーの全リフレッシュトークンを無効化
   */
  async revokeAllForUser(userId: string): Promise<void> {
    const redisClient = getRedisClient();
    if (!redisClient) {
      return;
    }

    const userKey = `${this.prefix}user:${userId}`;
    const tokenIds = await redisClient.sMembers(userKey);

    // 全トークンを削除
    const pipeline = redisClient.multi();
    for (const tokenId of tokenIds) {
      pipeline.del(`${this.prefix}${tokenId}`);
    }
    pipeline.del(userKey);
    await pipeline.exec();
  }

  /**
   * リフレッシュトークンの存在確認
   */
  async exists(tokenId: string): Promise<boolean> {
    const redisClient = getRedisClient();
    if (!redisClient) {
      return false;
    }

    const key = `${this.prefix}${tokenId}`;
    const exists = await redisClient.exists(key);
    return exists === 1;
  }
}
```

### Phase 4: トークン更新APIの実装

#### ファイル: `backend/src/controllers/authController.ts`（既存ファイルに追加）

```typescript
/**
 * トークン更新API
 * POST /api/auth/refresh
 */
export async function refreshToken(req: Request, res: Response): Promise<void> {
  try {
    // Cookieからリフレッシュトークンを取得
    const refreshToken = req.cookies.refreshToken;

    if (!refreshToken) {
      res.status(401).json({
        success: false,
        error: {
          code: 'REFRESH_TOKEN_MISSING',
          message: 'リフレッシュトークンが提供されていません'
        }
      });
      return;
    }

    // リフレッシュトークンを検証
    const jwtService = new JwtService();
    const verifyResult = await jwtService.verifyRefreshToken(refreshToken);

    if (!verifyResult.success) {
      res.status(401).json({
        success: false,
        error: verifyResult.error
      });
      return;
    }

    const payload = verifyResult.data;

    // Redisでトークンが有効かチェック
    const refreshTokenService = new RefreshTokenService();
    const exists = await refreshTokenService.exists(payload.tokenId);

    if (!exists) {
      res.status(401).json({
        success: false,
        error: {
          code: 'REFRESH_TOKEN_REVOKED',
          message: 'リフレッシュトークンは無効化されています'
        }
      });
      return;
    }

    // ユーザー情報を取得（データベースから）
    const userRepository = new UserRepository();
    const user = await userRepository.findById(payload.userId, payload.organizationId);

    if (!user) {
      res.status(401).json({
        success: false,
        error: {
          code: 'USER_NOT_FOUND',
          message: 'ユーザーが見つかりません'
        }
      });
      return;
    }

    // 権限情報を取得
    const permissionService = new PermissionService();
    const permissions = await permissionService.getUserPermissions(
      payload.userId,
      payload.organizationId
    );

    // 新しいトークンペアを生成
    const tokenPairResult = await jwtService.generateTokenPair({
      userId: payload.userId,
      organizationId: payload.organizationId,
      sessionId: payload.sessionId,
      permissions
    });

    if (!tokenPairResult.success) {
      res.status(500).json({
        success: false,
        error: tokenPairResult.error
      });
      return;
    }

    // 【トークンローテーション】古いリフレッシュトークンを無効化
    await refreshTokenService.revoke(payload.tokenId, payload.userId);

    // 新しいリフレッシュトークンをRedisに保存
    const newRefreshPayload = jwtService.decode(tokenPairResult.data.refreshToken);
    if (newRefreshPayload) {
      await refreshTokenService.save(
        newRefreshPayload as RefreshTokenPayload,
        req.ip,
        req.get('user-agent')
      );
    }

    // リフレッシュトークンをhttpOnly Cookieに設定
    res.cookie('refreshToken', tokenPairResult.data.refreshToken, {
      httpOnly: true,
      secure: process.env.NODE_ENV === 'production',
      sameSite: 'strict',
      maxAge: 7 * 24 * 60 * 60 * 1000, // 7日間
      path: '/api/auth/refresh'
    });

    // アクセストークンをレスポンスで返す
    res.json({
      success: true,
      data: {
        accessToken: tokenPairResult.data.accessToken,
        user: {
          id: user.user_id,
          username: user.user_name,
          email: user.email,
          roleName: user.role_name,
          organizationName: user.organization_name
        }
      }
    });
  } catch (error) {
    console.error('refreshToken error:', error);
    res.status(500).json({
      success: false,
      error: {
        code: 'SYSTEM_ERROR',
        message: 'システムエラーが発生しました'
      }
    });
  }
}
```

/**
 * ログアウトAPI
 * POST /api/auth/logout
 */
export async function logout(req: Request, res: Response): Promise<void> {
  try {
    const refreshToken = req.cookies.refreshToken;

    if (refreshToken) {
      // リフレッシュトークンをデコード
      const jwtService = new JwtService();
      const payload = jwtService.decode(refreshToken) as RefreshTokenPayload;

      if (payload && payload.tokenId) {
        // Redisからリフレッシュトークンを削除
        const refreshTokenService = new RefreshTokenService();
        await refreshTokenService.revoke(payload.tokenId, payload.userId);
      }

      // Cookieをクリア
      res.clearCookie('refreshToken', { path: '/api/auth/refresh' });
    }

    // セッションを破棄
    req.session.destroy((err) => {
      if (err) {
        console.error('Session destroy error:', err);
      }
    });

    res.json({
      success: true,
      data: {
        message: 'ログアウトしました'
      }
    });
  } catch (error) {
    console.error('logout error:', error);
    res.status(500).json({
      success: false,
      error: {
        code: 'SYSTEM_ERROR',
        message: 'システムエラーが発生しました'
      }
    });
  }
}

## 5. 技術的詳細

### トークン管理フロー

```
【ログイン時】
1. ユーザー認証
2. トークンペア生成
   - アクセストークン（15分）
   - リフレッシュトークン（7日間）
3. リフレッシュトークンをRedisに保存
4. リフレッシュトークンをhttpOnly Cookieに設定
5. アクセストークンをレスポンスで返す

【API呼び出し時】
1. アクセストークンをAuthorizationヘッダーに設定
2. authMiddlewareで検証
3. 有効期限切れの場合 → 401エラー
4. フロントエンドが自動的に /api/auth/refresh を呼び出し

【トークン更新時】
1. リフレッシュトークンをCookieから取得
2. JWT検証
3. Redis存在確認
4. 新しいトークンペア生成
5. 【トークンローテーション】古いリフレッシュトークンを無効化
6. 新しいリフレッシュトークンをRedisに保存
7. 新しいアクセストークンを返す

【ログアウト時】
1. リフレッシュトークンをRedisから削除
2. Cookieをクリア
3. セッション破棄
```

### Redis保存構造

```
Key形式:
  mmdpf:refresh_token:{tokenId}  → StoredRefreshToken（JSON）
  mmdpf:refresh_token:user:{userId}  → Set<tokenId>

TTL:
  自動的に有効期限で削除される
```

### セキュリティ考慮事項

#### 1. トークンローテーション
- リフレッシュトークン使用後、古いトークンを無効化
- 再利用攻撃を防止

#### 2. httpOnly Cookie
- JavaScriptからアクセス不可
- XSS攻撃対策

#### 3. 短命アクセストークン
- 15分で期限切れ
- 漏洩リスクの最小化

#### 4. Redis管理
- 中央集権的なトークン管理
- 即座の無効化が可能

#### 5. IPアドレス・User-Agent記録
- 不正アクセスの検知
- セキュリティ監査

## 6. リスクと対策

/**
 * アクセストークンを自動更新
 */
async function refreshAccessToken(): Promise<string | null> {
  try {
    const response = await fetch('/api/auth/refresh', {
      method: 'POST',
      credentials: 'include'  // Cookie送信
    });

    if (!response.ok) {
      // リフレッシュ失敗 → ログインページへ
      window.location.href = '/login';
      return null;
    }

    const data = await response.json();
    return data.data.accessToken;
  } catch (error) {
    console.error('Token refresh failed:', error);
    window.location.href = '/login';
    return null;
  }
}
