## 🎯 最終決定事項

### エディタ表示方式
❌ **モーダルダイアログ方式** → ✅ **専用ページ方式（画面遷移）**

### 理由
- 既存の編集画面（実験作業者など）と同じUXを提供
- より大きな作業スペースを確保
- ブラウザの戻るボタンで元の画面に戻れる
- URLでエディタ状態を管理可能

---

## 📋 画面遷移フロー

### 起動フロー
```
プロジェクト管理画面
  ↓
編集ボタンクリック
  ↓
Draft存在チェック
  ├─ YES → Draft復元確認ダイアログ
  │         ├─「復元する」→ /editor/[projectId]?mode=restore
  │         └─「新規/既存」→ /editor/[projectId]?mode=auto
  └─ NO → /editor/[projectId]?mode=auto
  ↓
エディタページ表示
  ├─ mode=auto: MAIMLファイル自動判定
  ├─ mode=restore: Draft復元
  └─ mode=create: 新規作成
```

### 戻るフロー
```
エディタページ
  ↓
「閉じる」ボタンまたはブラウザの戻るボタン
  ↓
未保存の変更がある？
  ├─ YES → 確認ダイアログ
  │         ├─「保存せずに戻る」→ プロジェクト管理画面
  │         └─「キャンセル」→ エディタページに留まる
  └─ NO → プロジェクト管理画面
```

### タスク1: エディタページ作成 ⭐⭐⭐

**ファイル**: `frontend/src/app/editor/[projectId]/page.tsx`

```typescript
'use client';

import React, { useState, useEffect, useCallback } from 'react';
import { useParams, useRouter, useSearchParams } from 'next/navigation';
import { Button, Spin, message } from 'antd';
import { ArrowLeftOutlined } from '@ant-design/icons';
import PetriNetEditor from '@/components/viewer/PetriNetEditor/PetriNetEditor';

/**
 * エディタページ
 * 
 * URL: /editor/[projectId]?mode=auto|restore|create
 */
export default function EditorPage() {
  const router = useRouter();
  const params = useParams();
  const searchParams = useSearchParams();
  
  const projectId = params.projectId as string;
  const mode = searchParams.get('mode') || 'auto';
  
  const [isLoading, setIsLoading] = useState(true);
  const [projectData, setProjectData] = useState<any>(null);
  const [maimlContent, setMaimlContent] = useState<string | undefined>(undefined);
  const [hasUnsavedChanges, setHasUnsavedChanges] = useState(false);

  // プロジェクト情報とファイルを読み込む
  useEffect(() => {
    loadProjectData();
  }, [projectId, mode]);

  const loadProjectData = async () => {
    try {
      setIsLoading(true);
      
      // プロジェクト情報を取得
      const projectResponse = await fetch(`/api/projects/${projectId}`);
      const project = await projectResponse.json();
      setProjectData(project);
      
      if (mode === 'restore') {
        // Draft復元モード
        const draftKey = `petri-net-draft-${projectId}`;
        const draftData = localStorage.getItem(draftKey);
        if (draftData) {
          setMaimlContent(draftData);
        }
      } else if (mode === 'auto' || mode === 'create') {
        // 自動判定または新規作成
        const filesResponse = await fetch(`/api/projects/${projectId}/files`);
        const files = await filesResponse.json();
        
        const maimlFiles = files.filter((f: any) => 
          f.name.endsWith('.maiml') && !f.isFolder
        );
        
        if (maimlFiles.length > 0 && mode === 'auto') {
          // 既存ファイルを読み込み
          const latestFile = maimlFiles.sort((a: any, b: any) => 
            new Date(b.lastModified).getTime() - new Date(a.lastModified).getTime()
          )[0];
          
          const contentResponse = await fetch(latestFile.path);
          const content = await contentResponse.text();
          setMaimlContent(content);
        } else {
          // 新規作成
          setMaimlContent(undefined);
        }
      }
    } catch (error) {
      console.error('Failed to load project data:', error);
      message.error('プロジェクトの読み込みに失敗しました');
    } finally {
      setIsLoading(false);
    }
  };

  // 戻るボタン
  const handleBack = useCallback(() => {
    if (hasUnsavedChanges) {
      // 未保存の変更がある場合
      if (window.confirm('保存されていない変更があります。破棄して戻りますか？')) {
        router.back();
      }
    } else {
      router.back();
    }
  }, [hasUnsavedChanges, router]);

  // 保存処理
  const handleSave = useCallback(async (content: string) => {
    // 保存処理は ProjectManagement の仕様に従う
    // （詳細は20251223_1020_Phase6_保存処理仕様_確定版.md参照）
    
    try {
      // TODO: API実装
      message.success('保存しました');
      setHasUnsavedChanges(false);
    } catch (error) {
      message.error('保存に失敗しました');
      throw error;
    }
  }, [projectId]);

  if (isLoading) {
    return (
      <div style={{ 
        display: 'flex', 
        justifyContent: 'center', 
        alignItems: 'center', 
        height: '100vh' 
      }}>
        <Spin size="large" tip="読み込み中..." />
      </div>
    );
  }

  return (
    <div style={{ 
      width: '100vw', 
      height: '100vh', 
      display: 'flex', 
      flexDirection: 'column' 
    }}>
      {/* ヘッダー */}
      <div style={{ 
        padding: '16px', 
        borderBottom: '1px solid #f0f0f0',
        display: 'flex',
        alignItems: 'center',
        gap: '16px'
      }}>
        <Button 
          icon={<ArrowLeftOutlined />} 
          onClick={handleBack}
        >
          戻る
        </Button>
        <h1 style={{ margin: 0, fontSize: '18px' }}>
          実験フロー編集 - {projectData?.name}
        </h1>
      </div>

      {/* エディタ本体 */}
      <div style={{ flex: 1, overflow: 'hidden' }}>
        <PetriNetEditor
          initialContent={maimlContent}
          projectId={projectId}
          onSave={handleSave}
          onContentChange={() => setHasUnsavedChanges(true)}
        />
      </div>
    </div>
  );
}
```

```typescript
'use client';

import React from 'react';
import { Modal, Button, Space, Alert, Typography } from 'antd';
import { InfoCircleOutlined } from '@ant-design/icons';

const { Text } = Typography;

interface DraftRestoreDialogProps {
  open: boolean;
  onClose: () => void;
  onRestoreDraft: () => void;
  onCreateNew: () => void;
  draftInfo: {
    savedAt: string;
    methodCount: number;
    nodeCount: number;
  };
}

/**
 * Draft復元確認ダイアログ
 * 
 * 一時保存データがある場合に表示し、復元するか新規作成するかを選択
 */
const DraftRestoreDialog: React.FC<DraftRestoreDialogProps> = ({
  open,
  onClose,
  onRestoreDraft,
  onCreateNew,
  draftInfo
}) => {
  return (
    <Modal
      title={
        <Space>
          <InfoCircleOutlined style={{ color: '#1890ff' }} />
          <span>一時保存データの復元</span>
        </Space>
      }
      open={open}
      onCancel={onClose}
      footer={[
        <Button key="new" onClick={onCreateNew}>
          新規作成/既存ファイルを開く
        </Button>,
        <Button key="restore" type="primary" onClick={onRestoreDraft}>
          一時保存を復元する
        </Button>
      ]}
      width={500}
    >
      <Space direction="vertical" size="middle" style={{ width: '100%' }}>
        <Alert
          message="一時保存データが見つかりました"
          description={`最終保存: ${new Date(draftInfo.savedAt).toLocaleString('ja-JP')}`}
          type="info"
          showIcon
        />
        
        <div style={{ 
          padding: '16px', 
          background: '#f5f5f5', 
          borderRadius: '4px' 
        }}>
          <Space direction="vertical" size="small">
            <Text>
              <strong>Method数:</strong> {draftInfo.methodCount}
            </Text>
            <Text>
              <strong>ノード数:</strong> {draftInfo.nodeCount}
            </Text>
          </Space>
        </div>

        <Alert
          message="注意"
          description="一時保存を復元しない場合、最新の保存済みファイルまたは新規作成から開始します。"
          type="warning"
          showIcon
        />
      </Space>
    </Modal>
  );
};

export default DraftRestoreDialog;
```

**編集ボタンのonClick実装:**
```typescript
const handleEditButtonClick = useCallback(async () => {
  if (!selectedProject) {
    message.warning('プロジェクトを選択してください');
    return;
  }

  // Draft チェック
  const draftKey = `petri-net-draft-${selectedProject.id}`;
  const draftData = localStorage.getItem(draftKey);
  
  if (draftData) {
    try {
      const draft = JSON.parse(draftData);
      const draftInfo = {
        savedAt: draft.savedAt || new Date().toISOString(),
        methodCount: draft.methods?.length || 0,
        nodeCount: draft.nodes?.length || 0
      };
      
      setDraftInfo(draftInfo);
      setDraftRestoreDialogOpen(true);
      return;
    } catch (error) {
      console.error('Draft parse error:', error);
    }
  }

  // Draftがない場合は直接エディタページへ
  router.push(`/editor/${selectedProject.id}?mode=auto`);
}, [selectedProject, router]);

// Draft復元ハンドラー
const handleRestoreDraft = useCallback(() => {
  setDraftRestoreDialogOpen(false);
  router.push(`/editor/${selectedProject?.id}?mode=restore`);
}, [selectedProject, router]);

// 新規作成/既存ファイルハンドラー
const handleCreateNewOrExisting = useCallback(() => {
  setDraftRestoreDialogOpen(false);
  router.push(`/editor/${selectedProject?.id}?mode=auto`);
}, [selectedProject, router]);
```
