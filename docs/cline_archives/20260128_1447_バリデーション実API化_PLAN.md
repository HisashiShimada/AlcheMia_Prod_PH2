```typescript
/**
 * MAIMLバリデーション
 */
export const validateMaiml = async (req: Request, res: Response) => {
  try {
    const { content } = req.body;
    const organizationId = req.organizationId;
    
    // バリデーション
    if (!content) {
      return res.status(400).json({
        success: false,
        error: {
          code: 'VALIDATION_ERROR',
          message: 'Content is required'
        }
      });
    }
    
    // MAIMLバリデーション実行
    const result = await maimlService.validateMaiml(content, organizationId);
    
    res.json({
      success: true,
      data: result,
      meta: {
        timestamp: new Date().toISOString(),
        requestId: req.headers['x-request-id'] || uuidv4()
      }
    });
    
  } catch (error) {
    console.error('❌ [maimlController] バリデーションエラー:', error);
    res.status(500).json({
      success: false,
      error: {
        code: 'INTERNAL_SERVER_ERROR',
        message: 'Validation failed',
        details: error.message
      }
    });
  }
};
```

**ファイル**: `backend/src/services/maimlService.ts`

```typescript
import { parseStringPromise } from 'xml2js';

/**
 * MAIMLバリデーション結果の型定義
 */
interface ValidationResult {
  valid: boolean;
  errors: ValidationError[];
  warnings: any[];
  corrections: any[];
}

interface ValidationError {
  type: string;
  message: string;
  description: string;
  xpath: string;
  elementType?: 'node' | 'edge';
  elementId?: string;
  severity: 'critical' | 'warning' | 'info';
}

/**
 * MAIMLコンテンツをバリデーション
 */
export const validateMaiml = async (
  content: string,
  organizationId: string
): Promise<ValidationResult> => {
  const errors: ValidationError[] = [];
  const warnings: any[] = [];
  const corrections: any[] = [];
  
  try {
    // 1. XML構文チェック
    await parseStringPromise(content, { strict: true });
    
    // 2. MAIML構造チェック
    validateMaimlStructure(content, errors);
    
    // 3. ペトリネット構造チェック
    validatePetriNetStructure(content, errors);
    
    // 4. テンプレート構造チェック
    validateTemplateStructure(content, errors);
    
    // 5. Document構造チェック
    validateDocumentStructure(content, errors);
    
    const valid = errors.filter(e => e.severity === 'critical').length === 0;
    
    return {
      valid,
      errors,
      warnings,
      corrections
    };
    
  } catch (error) {
    // XML構文エラー
    errors.push({
      type: 'syntax_error',
      message: 'XML構文エラー',
      description: error.message,
      xpath: '/',
      severity: 'critical'
    });
    
    return {
      valid: false,
      errors,
      warnings,
      corrections
    };
  }
};

/**
 * MAIML構造バリデーション
 */
function validateMaimlStructure(content: string, errors: ValidationError[]) {
  // ルート要素チェック
  if (!/<maiml[^>]*>/.test(content)) {
    errors.push({
      type: 'structure_error',
      message: 'MAIML root element not found',
      description: 'MAIML要素が見つかりません',
      xpath: '/maiml',
      severity: 'critical'
    });
  }
  
  // Document要素チェック
  if (!/<document[^>]*>/.test(content)) {
    errors.push({
      type: 'structure_error',
      message: 'Document element not found',
      description: 'Document要素が見つかりません',
      xpath: '/maiml/document',
      severity: 'critical'
    });
  }
  
  // Protocol要素チェック
  if (!/<protocol[^>]*>/.test(content)) {
    errors.push({
      type: 'structure_error',
      message: 'Protocol element not found',
      description: 'Protocol要素が見つかりません',
      xpath: '/maiml/protocol',
      severity: 'critical'
    });
  }
}

/**
 * ペトリネット構造バリデーション
 */
function validatePetriNetStructure(content: string, errors: ValidationError[]) {
  // PNML要素チェック
  const pnmlMatch = content.match(/<pnml[^>]*>/);
  if (!pnmlMatch) {
    errors.push({
      type: 'structure_error',
      message: 'PNML element not found',
      description: 'PNML要素が見つかりません',
      xpath: '/maiml/protocol/method/pnml',
      severity: 'critical'
    });
    return;
  }
  
  // Place要素チェック
  const placeMatches = content.matchAll(/<place\s+([^>]*)>/g);
  for (const match of placeMatches) {
    const attrString = match[1];
    const idMatch = attrString.match(/id="([^"]*)"/);
    
    if (!idMatch) {
      errors.push({
        type: 'validation_error',
        message: 'Place without id',
        description: 'Place要素にid属性がありません',
        xpath: '/maiml/protocol/method/pnml/net/place',
        elementType: 'node',
        severity: 'critical'
      });
    }
  }
  
  // Transition要素チェック
  const transitionMatches = content.matchAll(/<transition\s+([^>]*)>/g);
  for (const match of transitionMatches) {
    const attrString = match[1];
    const idMatch = attrString.match(/id="([^"]*)"/);
    
    if (!idMatch) {
      errors.push({
        type: 'validation_error',
        message: 'Transition without id',
        description: 'Transition要素にid属性がありません',
        xpath: '/maiml/protocol/method/pnml/net/transition',
        elementType: 'node',
        severity: 'critical'
      });
    }
  }
  
  // Arc要素チェック
  const arcMatches = content.matchAll(/<arc\s+([^>]*)>/g);
  for (const match of arcMatches) {
    const attrString = match[1];
    const idMatch = attrString.match(/id="([^"]*)"/);
    const sourceMatch = attrString.match(/source="([^"]*)"/);
    const targetMatch = attrString.match(/target="([^"]*)"/);
    
    if (!idMatch) {
      errors.push({
        type: 'validation_error',
        message: 'Arc without id',
        description: 'Arc要素にid属性がありません',
        xpath: '/maiml/protocol/method/pnml/net/arc',
        elementType: 'edge',
        severity: 'critical'
      });
    }
    
    if (!sourceMatch) {
      errors.push({
        type: 'validation_error',
        message: 'Arc without source',
        description: 'Arc要素にsource属性がありません',
        xpath: '/maiml/protocol/method/pnml/net/arc',
        elementType: 'edge',
        elementId: idMatch ? idMatch[1] : undefined,
        severity: 'critical'
      });
    }
    
    if (!targetMatch) {
      errors.push({
        type: 'validation_error',
        message: 'Arc without target',
        description: 'Arc要素にtarget属性がありません',
        xpath: '/maiml/protocol/method/pnml/net/arc',
        elementType: 'edge',
        elementId: idMatch ? idMatch[1] : undefined,
        severity: 'critical'
      });
    }
  }
}

/**
 * テンプレート構造バリデーション
 */
function validateTemplateStructure(content: string, errors: ValidationError[]) {
  // Material, Condition, Result テンプレートチェック
  const templateTypes = ['material', 'condition', 'result'];
  
  for (const type of templateTypes) {
    const templateMatches = content.matchAll(new RegExp(`<${type}\\s+([^>]*)>`, 'g'));
    
    for (const match of templateMatches) {
      const attrString = match[1];
      const idMatch = attrString.match(/id="([^"]*)"/);
      
      if (!idMatch) {
        errors.push({
          type: 'validation_error',
          message: `${type} template without id`,
          description: `${type}テンプレートにid属性がありません`,
          xpath: `/maiml/protocol/method/pnml/net/${type}`,
          severity: 'warning'
        });
      }
    }
  }
}

/**
 * Document構造バリデーション
 */
function validateDocumentStructure(content: string, errors: ValidationError[]) {
  // Document要素の必須子要素チェック
  const documentMatch = content.match(/<document[^>]*>([\s\S]*?)<\/document>/);
  
  if (documentMatch) {
    const documentContent = documentMatch[1];
    
    // name要素チェック
    if (!/<name[^>]*>/.test(documentContent)) {
      errors.push({
        type: 'validation_error',
        message: 'Document name not found',
        description: 'Document要素にname子要素がありません',
        xpath: '/maiml/document/name',
        severity: 'warning'
      });
    }
    
    // date要素チェック
    if (!/<date[^>]*>/.test(documentContent)) {
      errors.push({
        type: 'validation_error',
        message: 'Document date not found',
        description: 'Document要素にdate子要素がありません',
        xpath: '/maiml/document/date',
        severity: 'info'
      });
    }
  }
}
```

```typescript
/**
 * MAIMLバリデーション
 */
export const validateMaimlContent = async (content: string): Promise<any> => {
  try {
    const response = await fetch('/api/v1/maiml/validate', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
      },
      credentials: 'include',
      body: JSON.stringify({ content }),
    });
    
    const result = await response.json();
    
    if (!result.success) {
      throw new Error(result.error?.message || 'Validation failed');
    }
    
    return result.data;
    
  } catch (error) {
    console.error('❌ [apiClient] バリデーションエラー:', error);
    throw error;
  }
};
```

```typescript
const handleValidate = useCallback(async () => {
  try {
    setIsValidating(true);

        message: 'バリデーションシステムエラー',

    setIsValidating(false);
  }
```
