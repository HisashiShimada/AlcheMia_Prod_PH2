## 5. 実装詳細

### permissionCheckMiddleware.ts の設計

```typescript
/**
 * 権限チェックミドルウェア
 * 
 * @description
 * req.user.permissionsをチェックして、指定された機能への
 * アクセス権限を検証します
 */

import { Request, Response, NextFunction } from 'express';

/**
 * 権限チェックミドルウェアのファクトリー関数
 * 
 * @param functionId - チェック対象の機能ID（例: 'PRJ001'）
 * @param requiredPermission - 必要な権限レベル（0=書込, 1=読取）
 * @returns Expressミドルウェア関数
 * 
 * @example
 * // 読取権限が必要な場合
 * router.get('/api/projects', 
 *   authMiddleware, 
 *   checkPermission('PRJ001', 1), 
 *   controller.list
 * );
 * 
 * // 書込権限が必要な場合
 * router.post('/api/projects', 
 *   authMiddleware, 
 *   checkPermission('PRJ002', 0), 
 *   controller.create
 * );
 */
export function checkPermission(
  functionId: string,
  requiredPermission: number = 1  // デフォルトは読取権限
) {
  return (req: Request, res: Response, next: NextFunction): void => {
    try {
      // authMiddlewareで設定されたreq.userを確認
      if (!req.user) {
        res.status(401).json({
          success: false,
          error: {
            code: 'AUTH_REQUIRED',
            message: '認証が必要です'
          },
          meta: {
            timestamp: new Date().toISOString()
          }
        });
        return;
      }

      // permissions配列から該当するfunctionIdを検索
      const permission = req.user.permissions.find(
        p => p.functionId === functionId
      );

      // 権限が見つからない場合
      if (!permission) {
        res.status(403).json({
          success: false,
          error: {
            code: 'PERMISSION_DENIED',
            message: 'この機能へのアクセス権限がありません',
            details: {
              functionId,
              reason: '権限が設定されていません'
            }
          },
          meta: {
            timestamp: new Date().toISOString()
          }
        });
        return;
      }

      // 権限レベルのチェック
      // permission: 0=書込専用, 1=読取権限
      // 読取権限(1)があれば書込(0)もOK（読取の方が権限が強い）
      const hasPermission = permission.permission >= requiredPermission;

      if (!hasPermission) {
        res.status(403).json({
          success: false,
          error: {
            code: 'INSUFFICIENT_PERMISSION',
            message: '権限が不足しています',
            details: {
              functionId,
              required: requiredPermission === 0 ? '書込権限' : '読取権限',
              actual: permission.permission === 0 ? '書込専用' : '読取権限',
              reason: '必要な権限レベルに達していません'
            }
          },
          meta: {
            timestamp: new Date().toISOString()
          }
        });
        return;
      }

      // 権限チェック成功 - 次のミドルウェアへ
      next();
    } catch (error) {
      console.error('permissionCheckMiddleware error:', error);
      res.status(500).json({
        success: false,
        error: {
          code: 'SYSTEM_ERROR',
          message: 'システムエラーが発生しました'
        },
        meta: {
          timestamp: new Date().toISOString()
        }
      });
    }
  };
}

/**
 * 複数の機能IDをOR条件でチェックするミドルウェア
 * いずれかの機能に対して権限があればOK
 * 
 * @param functionIds - チェック対象の機能ID配列
 * @param requiredPermission - 必要な権限レベル
 */
export function checkAnyPermission(
  functionIds: string[],
  requiredPermission: number = 1
) {
  return (req: Request, res: Response, next: NextFunction): void => {
    try {
      if (!req.user) {
        res.status(401).json({
          success: false,
          error: {
            code: 'AUTH_REQUIRED',
            message: '認証が必要です'
          },
          meta: {
            timestamp: new Date().toISOString()
          }
        });
        return;
      }

      // いずれかの機能に権限があるかチェック
      const hasAnyPermission = functionIds.some(functionId => {
        const permission = req.user!.permissions.find(
          p => p.functionId === functionId
        );
        return permission && permission.permission >= requiredPermission;
      });

      if (!hasAnyPermission) {
        res.status(403).json({
          success: false,
          error: {
            code: 'PERMISSION_DENIED',
            message: 'この機能へのアクセス権限がありません',
            details: {
              functionIds,
              reason: 'いずれの権限も保持していません'
            }
          },
          meta: {
            timestamp: new Date().toISOString()
          }
        });
        return;
      }

      next();
    } catch (error) {
      console.error('checkAnyPermission error:', error);
      res.status(500).json({
        success: false,
        error: {
          code: 'SYSTEM_ERROR',
          message: 'システムエラーが発生しました'
        },
        meta: {
          timestamp: new Date().toISOString()
        }
      });
    }
  };
}

/**
 * 複数の機能IDをAND条件でチェックするミドルウェア
 * すべての機能に対して権限が必要
 * 
 * @param functionIds - チェック対象の機能ID配列
 * @param requiredPermission - 必要な権限レベル
 */
export function checkAllPermissions(
  functionIds: string[],
  requiredPermission: number = 1
) {
  return (req: Request, res: Response, next: NextFunction): void => {
    try {
      if (!req.user) {
        res.status(401).json({
          success: false,
          error: {
            code: 'AUTH_REQUIRED',
            message: '認証が必要です'
          },
          meta: {
            timestamp: new Date().toISOString()
          }
        });
        return;
      }

      // すべての機能に権限があるかチェック
      const missingPermissions = functionIds.filter(functionId => {
        const permission = req.user!.permissions.find(
          p => p.functionId === functionId
        );
        return !permission || permission.permission < requiredPermission;
      });

      if (missingPermissions.length > 0) {
        res.status(403).json({
          success: false,
          error: {
            code: 'INSUFFICIENT_PERMISSION',
            message: '一部の権限が不足しています',
            details: {
              missingFunctionIds: missingPermissions,
              reason: 'すべての権限が必要です'
            }
          },
          meta: {
            timestamp: new Date().toISOString()
          }
        });
        return;
      }

      next();
    } catch (error) {
      console.error('checkAllPermissions error:', error);
      res.status(500).json({
        success: false,
        error: {
          code: 'SYSTEM_ERROR',
          message: 'システムエラーが発生しました'
        },
        meta: {
          timestamp: new Date().toISOString()
        }
      });
    }
  };
}
```

### 使用例（ルート定義）

```typescript
// backend/src/routes/projectRoutes.ts の例

import { Router } from 'express';
import { authMiddleware } from '../middleware/authMiddleware';
import { checkPermission } from '../middleware/permissionCheckMiddleware';
import * as projectController from '../controllers/projectController';

const router = Router();

// プロジェクト一覧取得（読取権限必要）
router.get(
  '/projects',
  authMiddleware,                    // 1. JWT認証
  checkPermission('PRJ001', 1),      // 2. 読取権限チェック
  projectController.list
);

// プロジェクト作成（書込権限必要）
router.post(
  '/projects',
  authMiddleware,
  checkPermission('PRJ002', 0),      // 書込権限チェック
  projectController.create
);

// プロジェクト更新（書込権限必要）
router.put(
  '/projects/:id',
  authMiddleware,
  checkPermission('PRJ003', 0),
  projectController.update
);

// プロジェクト削除（書込権限必要）
router.delete(
  '/projects/:id',
  authMiddleware,
  checkPermission('PRJ004', 0),
  projectController.delete
);

export default router;
```

### 権限レベルの定義
```typescript
// permission値の意味
0: 書込専用権限（データの作成・更新・削除が可能）

// 権限の強さ: 1 > 0
// つまり、読取権限(1)を持っていれば書込(0)も可能
```
