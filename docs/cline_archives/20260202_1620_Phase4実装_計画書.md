## 📐 Phase 4実装計画

### Step 1: hasInstructionByTransitionId()ヘルパーメソッド追加

**新規メソッドの仕様:**

```typescript
/**
 * 指定されたTransitionに対するInstructionが存在するか判定
 * 
 * @param {string} methodId - Method ID
 * @param {string} transitionId - Transition ID
 * @returns {boolean} Instructionが存在する場合true
 * @private
 */
private hasInstructionByTransitionId(
  methodId: string,
  transitionId: string
): boolean {
  // 1. 該当するMethodを取得
  const method = this.getMethod(methodId);
  if (!method || !method.program) {
    return false;
  }
  
  // 2. program配列を取得
  const programs = Array.isArray(method.program) 
    ? method.program 
    : [method.program];
  
  // 3. 各programのinstructionを検索
  for (const program of programs) {
    if (!program.instruction) {
      continue;
    }
    
    const instructions = Array.isArray(program.instruction)
      ? program.instruction
      : [program.instruction];
    
    // 4. 該当するtransitionIdを持つinstructionを検索
    const hasMatch = instructions.some(
      instruction => instruction.ref === transitionId
    );
    
    if (hasMatch) {
      return true;
    }
  }
  
  return false;
}
```
