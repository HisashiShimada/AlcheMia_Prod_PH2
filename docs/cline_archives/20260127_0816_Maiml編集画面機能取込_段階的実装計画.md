##### Step 1.2: Export関数実装
**ファイル**: `frontend/src/utils/exportHelpers.ts` (新規作成)

```typescript
import { toPng, toSvg } from 'html-to-image';

/**
 * PNG形式でエクスポート
 */
export const exportToPng = async (elementSelector: string = '.react-flow') => {
  const element = document.querySelector(elementSelector) as HTMLElement;
  if (!element) {
    throw new Error('Export target element not found');
  }

  const dataUrl = await toPng(element, {
    backgroundColor: '#ffffff',
    quality: 1.0,
    pixelRatio: 2,
  });

  const link = document.createElement('a');
  link.download = `petri-net-${Date.now()}.png`;
  link.href = dataUrl;
  link.click();
};

/**
 * SVG形式でエクスポート
 */
export const exportToSvg = async (elementSelector: string = '.react-flow') => {
  const element = document.querySelector(elementSelector) as HTMLElement;
  if (!element) {
    throw new Error('Export target element not found');
  }

  const dataUrl = await toSvg(element, {
    backgroundColor: '#ffffff',
  });

  const link = document.createElement('a');
  link.download = `petri-net-${Date.now()}.svg`;
  link.href = dataUrl;
  link.click();
};

/**
 * JSON形式でエクスポート
 */
export const exportToJson = (nodes: any[], edges: any[], viewport: any) => {
  const flowData = {
    nodes,
    edges,
    viewport,
    metadata: {
      exportedAt: new Date().toISOString(),
      nodeCount: nodes.length,
      edgeCount: edges.length,
      version: '1.0.0'
    }
  };

  const dataStr = JSON.stringify(flowData, null, 2);
  const dataBlob = new Blob([dataStr], { type: 'application/json' });
  const url = URL.createObjectURL(dataBlob);

  const link = document.createElement('a');
  link.download = `petri-net-${Date.now()}.json`;
  link.href = url;
  link.click();

  URL.revokeObjectURL(url);
};
```

##### Step 1.3: EditorToolbarにExportボタン追加
**ファイル**: `frontend/src/components/editor/EditorToolbar.tsx`

```typescript
import { DownloadOutlined } from '@ant-design/icons';
import { Menu, Dropdown } from 'antd';
import { exportToPng, exportToSvg, exportToJson } from '@/utils/exportHelpers';

// EditorToolbar内に追加
const handleExport = async (format: 'png' | 'svg' | 'json') => {
  try {
    if (format === 'png') {
      await exportToPng();
    } else if (format === 'svg') {
      await exportToSvg();
    } else if (format === 'json') {
      const { nodes, edges } = getNodesAndEdges(); // 既存のReactFlow取得関数
      const viewport = getViewport(); // 既存のViewport取得関数
      exportToJson(nodes, edges, viewport);
    }
  } catch (error) {
    console.error('Export failed:', error);
    message.error('エクスポートに失敗しました');
  }
};

const exportMenu = (
  <Menu>
    <Menu.Item key="png" onClick={() => handleExport('png')}>
      PNG画像として保存
    </Menu.Item>
    <Menu.Item key="svg" onClick={() => handleExport('svg')}>
      SVG形式として保存
    </Menu.Item>
    <Menu.Item key="json" onClick={() => handleExport('json')}>
      JSON形式として保存
    </Menu.Item>
  </Menu>
);

// ツールバーに追加
<Dropdown overlay={exportMenu}>
  <Button icon={<DownloadOutlined />}>
    エクスポート
  </Button>
</Dropdown>
```
