## 🎯 実施内容

### Property型の統一化

**Before（細分化型）:**
```typescript
// 個別の型定義（24種類）
export interface StringProperty extends PropertyBase {
  'xsi:type': 'stringType';
  value?: string;
}

export interface IntProperty extends PropertyBase {
  'xsi:type': 'intType';
  value?: number;
  units?: string;
  formatString?: string;
  scaleFactor?: number;
}

export interface DoubleProperty extends PropertyBase {
  'xsi:type': 'doubleType';
  value?: number;
  units?: string;
  formatString?: string;
  scaleFactor?: number;
}

// ... 他21種類の個別型定義

export type Property =
  | StringProperty
  | IntProperty
  | DoubleProperty
  | ... // 24種類のUnion型
```

**After（統一型）:**
```typescript
/**
 * Property統一型
 * 
 * すべてのPropertyタイプを包括する統一型定義。
 * xsi:type属性で型を判別し、適切なプロパティにアクセス可能。
 */
export interface Property {
  // 必須フィールド
  key: string;
  'xsi:type': PropertyType;
  
  // オプションフィールド
  description?: string;
  
  // 値プロパティ（型により使い分け）
  value?: string | number | boolean;        // 単一値用
  values?: (string | number | boolean)[];   // リスト用
  properties?: Property[];                  // ネスト構造用
  
  // 数値型オプション
  units?: string;
  formatString?: string;
  scaleFactor?: number;
}
```

## ✅ 統一型の利点

### 1. シンプルで保守しやすい

**Before:**
```typescript
// 24種類の個別型を管理する必要があった
export interface StringProperty { ... }
export interface IntProperty { ... }
export interface DoubleProperty { ... }
// ... 21種類続く
```

**After:**
```typescript
// 1つの統一型のみ
export interface Property { ... }
```

### 2. 柔軟な型判別

**xsi:type属性で型を判別:**
```typescript
function processProperty(prop: Property) {
  switch (prop['xsi:type']) {
    case 'stringType':
      // prop.value は string
      return prop.value;
    
    case 'intType':
    case 'doubleType':
      // prop.value は number
      // prop.units でオプション取得
      return `${prop.value} ${prop.units || ''}`;
    
    case 'propertyListType':
      // prop.properties でネスト構造にアクセス
      return prop.properties?.map(p => processProperty(p));
    
    case 'intListType':
      // prop.values で配列にアクセス
      return prop.values;
  }
}
```

## 🔄 パーサーでの使用

### 型安全な処理

```typescript
import type { Property } from '@/types/maiml-unified';

function extractPropertyValue(prop: Property): string | number | boolean | undefined {
  // xsi:typeで型を判別
  const type = prop['xsi:type'];
  
  // 単一値の場合
  if (prop.value !== undefined) {
    return prop.value;
  }
  
  // リスト値の場合
  if (prop.values !== undefined) {
    return prop.values;
  }
  
  // ネスト構造の場合
  if (prop.properties !== undefined) {
    return prop.properties.map(p => extractPropertyValue(p));
  }
  
  return undefined;
}

function formatProperty(prop: Property): string {
  const value = prop.value;
  
  // 数値型で単位がある場合
  if (typeof value === 'number' && prop.units) {
    return `${value} ${prop.units}`;
  }
  
  return String(value);
}
```

### パーサー実装

```typescript
import type { Property, MaimlRoot } from '@/types/maiml-unified';

class MaiMLParser {
  // XSD準拠型を直接返す
  getProperties(templateId: string): Property[] {
    // maiml-unified型から抽出
    return this.typedMaiml.protocol.method[0]
      .program[0].materialTemplate
      ?.find(t => t.id === templateId)
      ?.property || [];
  }
  
  // 拡張情報を追加する場合はインライン型で
  getPropertiesExtended(templateId: string): Array<Property & {
    displayValue: string;
    category: string;
  }> {
    const props = this.getProperties(templateId);
    return props.map(p => ({
      ...p,  // XSD準拠型をベースに
      displayValue: this.formatPropertyValue(p),
      category: this.categorizeProperty(p)
    }));
  }
}
```
